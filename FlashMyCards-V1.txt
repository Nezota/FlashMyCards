#[derive(Debug, Clone, Copy, PartialEq)]
enum DisplayMode {
    Windowed,
    Borderless,
    Fullscreen,
}

impl DisplayMode {
    fn label(&self) -> &'static str {
        match self {
            DisplayMode::Windowed => "Windowed (1280x720)",
            DisplayMode::Borderless => "Borderless (Native)",
            DisplayMode::Fullscreen => "Fullscreen",
        }
    }

    fn next(&self) -> Self {
        match self {
            DisplayMode::Windowed => DisplayMode::Borderless,
            DisplayMode::Borderless => DisplayMode::Fullscreen,
            DisplayMode::Fullscreen => DisplayMode::Windowed,
        }
    }

    fn apply(&self) {
        match self {
            DisplayMode::Windowed => {
                set_fullscreen(false);
                request_new_screen_size(1280.0, 720.0);
            }
            DisplayMode::Borderless => {
                set_fullscreen(false);
                let (w, h) = detect_monitor_resolution();
                request_new_screen_size(w as f32, h as f32);
            }
            DisplayMode::Fullscreen => {
                set_fullscreen(true);
            }
        }
    }
}

fn detect_monitor_resolution() -> (i32, i32) {
    #[cfg(target_os = "windows")]
    {
        unsafe extern "system" {
            fn GetSystemMetrics(nIndex: i32) -> i32;
        }
        unsafe {
            let w = GetSystemMetrics(0);
            let h = GetSystemMetrics(1);
            if w > 0 && h > 0 {
                return (w, h);
            }
        }
    }
    (1920, 1080)
}

impl GameSettings {
    fn get_hex(&self, slot: ColorSlot) -> u32 {
        match slot {
            ColorSlot::CursorDot => self.hex_cursor_dot,
            ColorSlot::CursorTrail => self.hex_cursor_trail,
            ColorSlot::Background => self.hex_bg,
            ColorSlot::MenuAccent => self.hex_accent,
            ColorSlot::TextPrimary => self.hex_text_primary,
            ColorSlot::TextMuted => self.hex_text_muted,
        }
    }

    fn set_hex(&mut self, slot: ColorSlot, hex: u32) {
        match slot {
            ColorSlot::CursorDot => self.hex_cursor_dot = hex,
            ColorSlot::CursorTrail => self.hex_cursor_trail = hex,
            ColorSlot::Background => self.hex_bg = hex,
            ColorSlot::MenuAccent => self.hex_accent = hex,
            ColorSlot::TextPrimary => self.hex_text_primary = hex,
            ColorSlot::TextMuted => self.hex_text_muted = hex,
        }
    }

    fn cycle_preset(&mut self, slot: ColorSlot) {
        let current = self.get_hex(slot);
        let next_idx = COLOR_PRESETS
            .iter()
            .position(|&c| c == current)
            .map(|i| (i + 1) % COLOR_PRESETS.len())
            .unwrap_or(0);
        self.set_hex(slot, COLOR_PRESETS[next_idx]);
    }
}

#[derive(Debug, Clone, Copy, PartialEq)]
enum ContentType {
    Audio,
    Text,
    Image,
}

impl ContentType {
    fn from_str(s: &str) -> Self {
        match s.trim().to_ascii_lowercase().as_str() {
            "audio" | "sound" | "mp3" | "wav" | "ogg" => ContentType::Audio,
            "image" | "img" | "png" | "jpg" | "jpeg" => ContentType::Image,
            _ => ContentType::Text,
        }
    }
}

#[derive(Debug, Clone)]
struct CardItem {
    kind: ContentType,
    label: String,
}

impl CardItem {
    fn display_text(&self) -> String {
        if self.kind == ContentType::Text {
            self.label.clone()
        } else {
            Path::new(&self.label)
                .file_name()
                .and_then(|n| n.to_str())
                .unwrap_or(&self.label)
                .to_string()
        }
    }
}

fn parse_csv_row(line: &str) -> Vec<String> {
    let mut fields = Vec::new();
    let mut current = String::new();
    let mut in_quotes = false;
    let mut chars = line.chars().peekable();

    while let Some(c) = chars.next() {
        if c == '"' {
            if in_quotes && chars.peek() == Some(&'"') {
                current.push('"');
                chars.next();
            } else {
                in_quotes = !in_quotes;
            }
        } else if c == ',' && !in_quotes {
            fields.push(current.trim().to_string());
            current.clear();
        } else {
            current.push(c);
        }
    }
    fields.push(current.trim().to_string());
    fields
}

fn load_deck_from_csv(csv_path: &Path) -> Result<Vec<FlashCard>, String> {
    let raw = fs::read_to_string(csv_path)
        .map_err(|e| format!("Could not read {}: {}", csv_path.display(), e))?;

    let mut cards = Vec::new();
    for (line_idx, raw_line) in raw.lines().enumerate() {
        let trimmed = raw_line.trim();
        if trimmed.is_empty() || trimmed.starts_with('#') {
            continue;
        }
        let cols = parse_csv_row(trimmed);
        if cols.len() < 4 {
            continue;
        }

        if line_idx == 0
            && cols[0].to_ascii_lowercase().contains("question")
            && cols[2].to_ascii_lowercase().contains("answer")
        {
            continue;
        }

        let q_type_str = cols.get(0).map(|s| s.as_str()).unwrap_or("");
        let q_val = cols.get(1).map(|s| s.trim()).unwrap_or("");
        let a1_type_str = cols.get(2).map(|s| s.as_str()).unwrap_or("");
        let a1_val = cols.get(3).map(|s| s.trim()).unwrap_or("");

        if q_val.is_empty() || a1_val.is_empty() {
            continue;
        }

        let question = CardItem {
            kind: ContentType::from_str(q_type_str),
            label: q_val.to_string(),
        };
        let answer = CardItem {
            kind: ContentType::from_str(a1_type_str),
            label: a1_val.to_string(),
        };

        let mut extra_answers = Vec::new();
        for pair_idx in [4usize, 6usize] {
            if let (Some(t), Some(v)) = (cols.get(pair_idx), cols.get(pair_idx + 1)) {
                if !v.trim().is_empty() {
                    extra_answers.push(CardItem {
                        kind: ContentType::from_str(t),
                        label: v.trim().to_string(),
                    });
                }
            }
        }

        let mut hints = Vec::new();
        for hint_col in [8usize, 10usize, 12usize] {
            if let (Some(t), Some(v)) = (cols.get(hint_col), cols.get(hint_col + 1)) {
                if !v.trim().is_empty() {
                    hints.push(CardItem {
                        kind: ContentType::from_str(t),
                        label: v.trim().to_string(),
                    });
                }
            }
        }

        cards.push(FlashCard {
            id: cards.len(),
            question,
            answer,
            extra_answers,
            hints,
        });
    }

    if cards.is_empty() {
        Err("No valid flashcard rows found in CSV.".to_string())
    } else {
        Ok(cards)
    }
}

#[derive(Debug, Clone)]
struct FloatDifficulty {
    answer_time: f32,
    num_answers: usize,
    round_time: f32,
    infinite_round: bool,
}

impl Default for FloatDifficulty {
    fn default() -> Self {
        Self {
            answer_time: 12.0,
            num_answers: 4,
            round_time: 60.0,
            infinite_round: false,
        }
    }
}

impl FloatDifficulty {
    fn is_highscore_eligible(&self) -> bool {

#[derive(Debug, Clone)]
struct FloatingTarget {
    card_id: usize,
    item: CardItem,
    pos: Vec2,
    vel: Vec2,
    spawn_scale: f32,
    hovered_audio: bool,
}

impl FloatingTarget {
    fn half_extents(&self) -> Vec2 {
        match self.item.kind {
            ContentType::Audio | ContentType::Image => {
                vec2(50.0 * self.spawn_scale, 50.0 * self.spawn_scale)
            }
            ContentType::Text => vec2(95.0 * self.spawn_scale, 38.0 * self.spawn_scale),
        }
    }

    fn hit_radius(&self) -> f32 {
        match self.item.kind {
            ContentType::Audio | ContentType::Image => 50.0 * self.spawn_scale,
            ContentType::Text => 70.0 * self.spawn_scale,
        }
    }

    fn contains(&self, point: Vec2) -> bool {
        if self.spawn_scale < 0.25 {
            return false;
        }
        match self.item.kind {
            ContentType::Audio | ContentType::Image => {
                (point - self.pos).length() <= 50.0 * self.spawn_scale
            }
            ContentType::Text => {
                let w = 190.0 * self.spawn_scale;
                let h = 76.0 * self.spawn_scale;
                let rect = Rect::new(self.pos.x - w / 2.0, self.pos.y - h / 2.0, w, h);
                rect.contains(point)
            }
        }
    }
}

#[derive(Debug, Clone)]
struct TileTarget {
    grid_row: usize,
    grid_col: usize,
    card_id: usize,
    item: CardItem,
    spawn_scale: f32,
    hovered_audio: bool,
}

impl TileTarget {
    fn rect(&self, grid_origin: Vec2, cell_size: f32) -> Rect {
        let pad = 6.0;
        let x = grid_origin.x + (self.grid_col as f32 * cell_size) + pad;
        let y = grid_origin.y + (self.grid_row as f32 * cell_size) + pad;
        let size = cell_size - pad * 2.0;
        let center = vec2(x + size / 2.0, y + size / 2.0);
        let current_size = size * self.spawn_scale;
        Rect::new(
            center.x - current_size / 2.0,
            center.y - current_size / 2.0,
            current_size,
            current_size,
        )
    }

    fn contains(&self, point: Vec2, grid_origin: Vec2, cell_size: f32) -> bool {
        if self.spawn_scale < 0.25 {
            return false;
        }
        self.rect(grid_origin, cell_size).contains(point)
    }
}

#[derive(Debug, Clone)]
struct DropTile {
    lane: usize,
    card_id: usize,
    item: CardItem,
    y: f32,
    hovered_audio: bool,
}

impl DropTile {
    fn rect(&self, playfield: Rect, num_lanes: usize) -> Rect {
        let lane_w = playfield.w / num_lanes as f32;
        let pad_x = (lane_w * 0.08).clamp(6.0, 16.0);
        let w = (lane_w - pad_x * 2.0).max(40.0);
        let h = 78.0;
        let x = playfield.x + (self.lane as f32 * lane_w) + pad_x;
        Rect::new(x, self.y, w, h)
    }

    fn contains(&self, point: Vec2, playfield: Rect, num_lanes: usize) -> bool {
        self.rect(playfield, num_lanes).contains(point)
    }
}

fn shuffle_slice<T>(slice: &mut [T]) {
    for i in (1..slice.len()).rev() {
        let j = rand::gen_range(0, i + 1);
        slice.swap(i, j);
    }
}

fn resolve_deck_asset_path(base_dir: Option<&Path>, raw_rel: &str) -> Option<String> {
    let clean = raw_rel.trim().trim_start_matches(['/', '\\']);
    if clean.is_empty() {
        return None;
    }

    let mut candidates = Vec::new();
    if let Some(dir) = base_dir {
        candidates.push(dir.join(clean));
    }
    candidates.push(PathBuf::from(DECKS_DIR).join(clean));
    candidates.push(PathBuf::from("assets").join(clean));
    candidates.push(PathBuf::from(clean));

    for cand in candidates {
        if cand.is_file() {
            return Some(cand.to_string_lossy().replace('\\', "/"));
        }
    }
    None
}

// TEXT WRAPPING & SMART SINGLE-WORD FITTING HELPERS

fn is_single_word_token(text: &str) -> bool {
    let trimmed = text.trim();
    !trimmed.is_empty() && !trimmed.chars().any(|c| c.is_whitespace())
}

fn truncate_with_ellipsis(text: &str, max_width: f32, font_size: u16, font: Option<&Font>) -> String {
    if measure_text(text, font, font_size, 1.0).width <= max_width {
        return text.to_string();
    }
    let ellipsis = "...";
    let mut out = String::new();
    for ch in text.chars() {
        let mut test = out.clone();
        test.push(ch);
        test.push_str(ellipsis);
        if measure_text(&test, font, font_size, 1.0).width > max_width {
            break;
        }
        out.push(ch);
    }
    if out.is_empty() {
        ellipsis.to_string()
    } else {
        format!("{}{}", out, ellipsis)
    }
}

fn wrap_text_lines(text: &str, max_width: f32, font_size: u16, font: Option<&Font>) -> Vec<String> {
    let mut lines = Vec::new();
    for raw_line in text.lines() {
        let words: Vec<&str> = raw_line.split_whitespace().collect();
        if words.is_empty() {
            let mut current = String::new();
            for ch in raw_line.chars() {
                let mut test = current.clone();
                test.push(ch);
                if !current.is_empty() && measure_text(&test, font, font_size, 1.0).width > max_width {
                    lines.push(current);
                    current = ch.to_string();
                } else {
                    current.push(ch);
                }
            }
            if !current.is_empty() {
                lines.push(current);
            }
            continue;
        }

        let mut current_line = String::new();
        for word in words {
            if measure_text(word, font, font_size, 1.0).width > max_width {
                if !current_line.is_empty() {
                    lines.push(current_line.clone());
                    current_line.clear();
                }
                for ch in word.chars() {
                    let mut test = current_line.clone();
                    test.push(ch);
                    if !current_line.is_empty()
                        && measure_text(&test, font, font_size, 1.0).width > max_width
                    {
                        lines.push(current_line);
                        current_line = ch.to_string();
                    } else {
                        current_line.push(ch);
                    }
                }
            } else {
                let candidate = if current_line.is_empty() {
                    word.to_string()
                } else {
                    format!("{} {}", current_line, word)
                };
                if measure_text(&candidate, font, font_size, 1.0).width <= max_width {
                    current_line = candidate;
                } else {
                    if !current_line.is_empty() {
                        lines.push(current_line);
                    }
                    current_line = word.to_string();
                }
            }
        }
        if !current_line.is_empty() {
            lines.push(current_line);
        }
    }
    if lines.is_empty() {
        lines.push(text.to_string());
    }
    lines
}

fn draw_fitted_wrapped_text(
    text: &str,
    rect: Rect,
    preferred_font: u16,
    min_font: u16,
    color: Color,
    font: Option<&Font>,
) {
    let pad_x = 10.0;
    let pad_y = 8.0;
    let max_w = (rect.w - pad_x * 2.0).max(16.0);
    let max_h = (rect.h - pad_y * 2.0).max(14.0);

    let trimmed = text.trim();
    if is_single_word_token(trimmed) {
        let single_min = min_font.min(MIN_SINGLE_WORD_FONT_SIZE);
        let mut try_font = preferred_font.max(single_min);
        while try_font >= single_min {
            let dims = measure_text(trimmed, font, try_font, 1.0);
            if dims.width <= max_w && (try_font as f32) <= max_h {
                let lx = rect.x + (rect.w - dims.width) * 0.5;
                let ly = rect.y + rect.h * 0.5 + try_font as f32 * 0.34;
                draw_ui_text(trimmed, lx, ly, try_font, color, font);
                return;
            }
            if try_font == single_min {
                break;
            }
            try_font -= 1;
        }
    }

    let mut chosen_font = preferred_font.max(min_font);
    let mut final_lines = Vec::new();

    while chosen_font >= min_font {
        let lines = wrap_text_lines(text, max_w, chosen_font, font);
        let line_height = chosen_font as f32 * 1.18;
        let total_h = lines.len() as f32 * line_height;
        if total_h <= max_h || chosen_font == min_font {
            final_lines = lines;
            break;
        }
        chosen_font -= 1;
    }

    let line_height = chosen_font as f32 * 1.18;
    let max_lines = ((max_h / line_height).floor() as usize).max(1);
    if final_lines.len() > max_lines {
        final_lines.truncate(max_lines);
        if let Some(last) = final_lines.last_mut() {
            *last = truncate_with_ellipsis(&format!("{}...", last), max_w, chosen_font, font);
        }
    }
    for line in &mut final_lines {
        if measure_text(&mut *line, font, chosen_font, 1.0).width > max_w {
            *line = truncate_with_ellipsis(line, max_w, chosen_font, font);
        }
    }

    let total_block_h = final_lines.len() as f32 * line_height;
    let start_y = rect.y + (rect.h - total_block_h) * 0.5 + chosen_font as f32 * 0.82;

    for (idx, line) in final_lines.iter().enumerate() {
        let dims = measure_text(line, font, chosen_font, 1.0);
        let lx = rect.x + (rect.w - dims.width) * 0.5;
        let ly = start_y + idx as f32 * line_height;
        draw_ui_text(line, lx, ly, chosen_font, color, font);
    }
}

fn load_combined_decks(selected_entries: &[DeckEntry]) -> (Vec<FlashCard>, String) {
    let mut combined_cards = Vec::new();
    let mut names = Vec::new();

    for entry in selected_entries {
        names.push(entry.name.clone());
        if let Some(path) = &entry.csv_path {
            if let Ok(mut cards) = load_deck_from_csv(path) {
                let base_dir = entry.base_dir.as_deref();
                for card in &mut cards {
                    card.id = combined_cards.len();
                    let mut items = vec![&mut card.question, &mut card.answer];
                    items.extend(card.extra_answers.iter_mut());
                    items.extend(card.hints.iter_mut());
                    for item in items {
                        if matches!(item.kind, ContentType::Image | ContentType::Audio) {
                            if let Some(resolved) = resolve_deck_asset_path(base_dir, &item.label) {
                                item.label = resolved;
                            }
                        }
                    }
                    combined_cards.push(card.clone());
                }
            }
        }
    }

    let combined_name = if names.is_empty() {
        "No Deck Selected".to_string()
    } else if names.len() == 1 {
        names[0].clone()
    } else {
        format!("{} (+{} more)", names[0], names.len() - 1)
    };

    (combined_cards, combined_name)
}

impl FloatSession {
    async fn new(selected_entries: &[DeckEntry], difficulty: FloatDifficulty, sounds: &SoundAssets) -> Self {
        let (deck, combined_deck_name) = load_combined_decks(selected_entries);

        let mut textures = HashMap::new();
        let mut audio_clips = HashMap::new();

        for card in &deck {
            let mut items = vec![&card.question, &card.answer];
            items.extend(card.extra_answers.iter());
            items.extend(card.hints.iter());

            for item in items {
                match item.kind {
                    ContentType::Image => {
                        if !textures.contains_key(&item.label) {
                            let resolved = resolve_deck_asset_path(None, &item.label)
                                .unwrap_or_else(|| item.label.clone());
                            if let Ok(tex) = load_texture(&resolved).await {
                                textures.insert(item.label.clone(), tex);
                            }
                        }
                    }
                    ContentType::Audio => {
                        if !audio_clips.contains_key(&item.label) {
                            let resolved = resolve_deck_asset_path(None, &item.label)
                                .unwrap_or_else(|| item.label.clone());
                            if let Ok(snd) = load_sound(&resolved).await {
                                audio_clips.insert(item.label.clone(), snd);
                            }
                        }
                    }
                    ContentType::Text => {}
                }
            }
        }

        let mut queue = VecDeque::new();
        let mut indices: Vec<usize> = (0..deck.len()).collect();
        shuffle_slice(&mut indices);
        for idx in indices {
            queue.push_back(QueuedQuestion {
                card_idx: idx,
                cohort_id: 0,
            });
        }

        let current_question = queue.pop_front().unwrap_or(QueuedQuestion {
            card_idx: 0,
            cohort_id: 0,
        });

        let q_time = difficulty.answer_time.max(0.5);
        let mut session = Self {
            deck_name: combined_deck_name,
            deck,
            difficulty: difficulty.clone(),
            textures,
            audio_clips,
            currently_playing_clip: None,
            queue,
            current_question,
            next_cohort_id: 1,
            loops_completed: 0,
            round_time_left: difficulty.round_time,
            question_time_left: q_time,
            score: 0,
            hints_revealed: [false; 3],
            audio_replay_pulse: 0.0,
            hint_audio_pulse: [0.0; 3],
            targets: Vec::new(),
            pops: Vec::new(),
            wrong_answer_triggered: false,
        };

        session.ensure_queue_refilled();
        session.populate_initial_targets();

        if !session.deck.is_empty()
            && session.deck[session.current_question.card_idx].question.kind == ContentType::Audio
        {
            let label = session.deck[session.current_question.card_idx]
                .question
                .label
                .clone();
            session.play_item_audio(&label, sounds);
            session.audio_replay_pulse = 0.6;
        }

        session
    }

    fn stop_active_clip(&mut self, sounds: &SoundAssets) {
        if let Some(prev_key) = self.currently_playing_clip.take() {
            if let Some(snd) = self.audio_clips.get(&prev_key) {
                stop_sound(snd);
            }
        }
        if let Some(s) = &sounds.placeholder_audio {
            stop_sound(s);
        }
    }

    fn play_item_audio(&mut self, label: &str, sounds: &SoundAssets) {
        self.stop_active_clip(sounds);
        if let Some(snd) = self.audio_clips.get(label) {
            play_sound_once(snd);
            self.currently_playing_clip = Some(label.to_string());
        } else if let Some(s) = &sounds.placeholder_audio {
            play_sound_once(s);
        }
    }

    fn ensure_queue_refilled(&mut self) {
        while self.queue.len() < 15 && !self.deck.is_empty() {
            let mut indices: Vec<usize> = (0..self.deck.len()).collect();
            shuffle_slice(&mut indices);
            let cohort = self.next_cohort_id;
            self.next_cohort_id += 1;
            for idx in indices {
                self.queue.push_back(QueuedQuestion {
                    card_idx: idx,
                    cohort_id: cohort,
                });
            }
        }
    }

    fn random_playfield_pos() -> Vec2 {
        let box_rect = float_playfield_rect();
        let pad_x = 105.0;
        let pad_y = 60.0;
        let min_x = box_rect.x + pad_x;
        let max_x = (box_rect.x + box_rect.w - pad_x).max(min_x + 10.0);
        let min_y = box_rect.y + pad_y;
        let max_y = (box_rect.y + box_rect.h - pad_y).max(min_y + 10.0);
        vec2(
            rand::gen_range(min_x, max_x),
            rand::gen_range(min_y, max_y),
        )
    }

    fn random_drift_velocity() -> Vec2 {
        let angle = rand::gen_range(0.0, std::f32::consts::TAU);
        let speed = rand::gen_range(3.0, 8.0);
        vec2(angle.cos() * speed, angle.sin() * speed)
    }

    fn spawn_target_for_card(&mut self, card_idx: usize) {
        let card = &self.deck[card_idx];
        self.targets.push(FloatingTarget {
            card_id: card.id,
            item: card.answer.clone(),
            pos: Self::random_playfield_pos(),
            vel: Self::random_drift_velocity(),
            spawn_scale: 0.01,
            hovered_audio: false,
        });
    }

    fn populate_initial_targets(&mut self) {
        self.targets.clear();
        self.wrong_answer_triggered = false;
        let count = self.difficulty.num_answers.clamp(2, 10);

        self.ensure_queue_refilled();

        let mut queued_card_indices = vec![self.current_question.card_idx];
        for q in &self.queue {
            if !queued_card_indices.contains(&q.card_idx) {
                queued_card_indices.push(q.card_idx);
            }
        }
        shuffle_slice(&mut queued_card_indices);

        let mut chosen = vec![self.current_question.card_idx];
        for &idx in &queued_card_indices {
            if idx != self.current_question.card_idx && chosen.len() < count {
                chosen.push(idx);
            }
        }

        while chosen.len() < count {
            let rnd = rand::gen_range(0, self.deck.len());
            if !chosen.contains(&rnd) {
                chosen.push(rnd);
            }
        }

        shuffle_slice(&mut chosen);
        for &idx in &chosen {
            self.spawn_target_for_card(idx);
        }
    }

    fn ensure_correct_answer_on_screen(&mut self) {
        if self.wrong_answer_triggered {
            return;
        }

        let correct_id = self.deck[self.current_question.card_idx].id;
        let has_correct = self.targets.iter().any(|t| t.card_id == correct_id);

        if !has_correct {
            if !self.targets.is_empty() {
                let replace_idx = rand::gen_range(0, self.targets.len());
                let old_pos = self.targets[replace_idx].pos;
                self.pops.push(PopEffect {
                    pos: old_pos,
                    base_radius: 48.0,
                    timer: 0.0,
                    duration: 0.28,
                    color: Color::from_hex(DEFAULT_HEX_TEXT_MUTED),
                });
                self.targets.remove(replace_idx);
            }
            self.spawn_target_for_card(self.current_question.card_idx);
        }

        let desired = self.difficulty.num_answers.clamp(2, 10);
        while self.targets.len() < desired {
            self.ensure_queue_refilled();
            let mut next_cand = self.current_question.card_idx;
            for q in &self.queue {
                if !self.targets.iter().any(|t| t.card_id == self.deck[q.card_idx].id) {
                    next_cand = q.card_idx;
                    break;
                }
            }
            self.spawn_target_for_card(next_cand);
        }
    }

    fn advance_question(
        &mut self,
        answered_correctly: bool,
        pop_pos: Vec2,
        pop_color: Color,
        sounds: &SoundAssets,
    ) {
        self.pops.push(PopEffect {
            pos: pop_pos,
            base_radius: 56.0,
            timer: 0.0,
            duration: 0.35,
            color: pop_color,
        });

        let finished_q = self.current_question;
        if !answered_correctly {
            let insert_pos = self
                .queue
                .iter()
                .rposition(|q| q.cohort_id == finished_q.cohort_id)
                .map(|idx| idx + 1)
                .unwrap_or(0);
            self.queue.insert(insert_pos, finished_q);
        } else {
            let cohort_still_remaining = self
                .queue
                .iter()
                .any(|q| q.cohort_id == finished_q.cohort_id);
            if !cohort_still_remaining {
                self.loops_completed += 1;
            }
        }

        self.ensure_queue_refilled();
        if let Some(next_q) = self.queue.pop_front() {
            self.current_question = next_q;
        }

        self.question_time_left = self.difficulty.answer_time.max(0.5);
        self.hints_revealed = [false; 3];
        self.hint_audio_pulse = [0.0; 3];
        self.wrong_answer_triggered = false;
        self.ensure_correct_answer_on_screen();

        if self.deck[self.current_question.card_idx].question.kind == ContentType::Audio {
            let label = self.deck[self.current_question.card_idx]
                .question
                .label
                .clone();
            self.play_item_audio(&label, sounds);
            self.audio_replay_pulse = 0.6;
        }
    }

    fn update(
        &mut self,
        dt: f32,
        mouse_pos: Vec2,
        settings: &GameSettings,
        sounds: &SoundAssets,
    ) -> bool {
        let accent = Color::from_hex(settings.hex_accent);

        if self.audio_replay_pulse > 0.0 {
            self.audio_replay_pulse = (self.audio_replay_pulse - dt).max(0.0);
        }
        for p in &mut self.hint_audio_pulse {
            if *p > 0.0 {
                *p = (*p - dt).max(0.0);
            }
        }

        let replay_triggered = is_mouse_button_pressed(MouseButton::Right)
            || is_key_pressed(KeyCode::LeftShift)
            || is_key_pressed(KeyCode::RightShift);

        if replay_triggered
            && self.deck[self.current_question.card_idx].question.kind == ContentType::Audio
        {
            self.audio_replay_pulse = 0.6;
            let label = self.deck[self.current_question.card_idx]
                .question
                .label
                .clone();
            self.play_item_audio(&label, sounds);
        }

        if !self.difficulty.infinite_round {
            self.round_time_left -= dt;
            if self.round_time_left <= 0.0 {
                self.round_time_left = 0.0;
                return true;
            }
        }

        let total_q_time = self.difficulty.answer_time.max(0.5);
        if !self.wrong_answer_triggered {
            self.question_time_left -= dt;
        }
        let ratio_left = (self.question_time_left / total_q_time).clamp(0.0, 1.0);

        let num_hints = self.deck[self.current_question.card_idx].hints.len().min(3);
        let thresholds = [0.75, 0.50, 0.25];
        for i in 0..num_hints {
            let key_pressed = is_key_pressed(settings.hint_keys[i]);
            let auto_reveal =
                !self.wrong_answer_triggered && !self.hints_revealed[i] && ratio_left <= thresholds[i];

            if auto_reveal {
                self.hints_revealed[i] = true;
                if self.deck[self.current_question.card_idx].hints[i].kind == ContentType::Audio {
                    let label = self.deck[self.current_question.card_idx].hints[i]
                        .label
                        .clone();
                    self.play_item_audio(&label, sounds);
                    self.hint_audio_pulse[i] = 0.75;
                }
            } else if key_pressed {
                self.hints_revealed[i] = true;
                if self.deck[self.current_question.card_idx].hints[i].kind == ContentType::Audio {
                    let label = self.deck[self.current_question.card_idx].hints[i]
                        .label
                        .clone();
                    self.play_item_audio(&label, sounds);
                    self.hint_audio_pulse[i] = 0.75;
                }
            }
        }

        if is_key_pressed(KeyCode::Space) && !self.wrong_answer_triggered {
            if let Some(s) = &sounds.skipped {
                play_sound_once(s);
            }
            let q_pos = vec2(screen_width() / 2.0, 100.0);
            self.advance_question(false, q_pos, Color::from_hex(0xFB7185), sounds);
            return false;
        }

        if self.question_time_left <= 0.0 && !self.wrong_answer_triggered {
            if let Some(s) = &sounds.missed {
                play_sound_once(s);
            }
            let q_pos = vec2(screen_width() / 2.0, 100.0);
            self.advance_question(false, q_pos, Color::from_hex(0xFB7185), sounds);
            return false;
        }

        let box_rect = float_playfield_rect();
        for target in &mut self.targets {
            if target.spawn_scale < 1.0 {
                target.spawn_scale = (target.spawn_scale + dt * 3.8).min(1.0);
            }

            let hovered = target.contains(mouse_pos);
            let target_speed = if hovered { 45.0 } else { 5.0 };
            let current_vel_len = target.vel.length();
            if current_vel_len > 0.1 {
                let dir = target.vel / current_vel_len;
                let new_len = current_vel_len + (target_speed - current_vel_len) * dt * 3.0;
                target.vel = dir * new_len;
            } else {
                target.vel =
                    vec2(rand::gen_range(-5.0, 5.0), rand::gen_range(-5.0, 5.0)).normalize() * 5.0;
            }

            target.pos += target.vel * dt;

            let ext = target.half_extents();
            let min_x = box_rect.x + ext.x + 6.0;
            let max_x = (box_rect.x + box_rect.w - ext.x - 6.0).max(min_x + 2.0);
            let min_y = box_rect.y + ext.y + 6.0;
            let max_y = (box_rect.y + box_rect.h - ext.y - 6.0).max(min_y + 2.0);

            if target.pos.x < min_x {
                target.pos.x = min_x;
                target.vel.x = target.vel.x.abs();
            } else if target.pos.x > max_x {
                target.pos.x = max_x;
                target.vel.x = -target.vel.x.abs();
            }

            if target.pos.y < min_y {
                target.pos.y = min_y;
                target.vel.y = target.vel.y.abs();
            } else if target.pos.y > max_y {
                target.pos.y = max_y;
                target.vel.y = -target.vel.y.abs();
            }
        }

        let len = self.targets.len();
        for i in 0..len {
            for j in (i + 1)..len {
                let delta = self.targets[j].pos - self.targets[i].pos;
                let dist = delta.length();
                let min_dist = self.targets[i].hit_radius() + self.targets[j].hit_radius() + 8.0;
                if dist > 0.001 && dist < min_dist {
                    let n = delta / dist;
                    let push = (min_dist - dist) * 0.5;
                    self.targets[i].pos -= n * push;
                    self.targets[j].pos += n * push;
                    let temp = self.targets[i].vel;
                    self.targets[i].vel = self.targets[j].vel;
                    self.targets[j].vel = temp;
                }
            }
        }

        for idx in 0..self.targets.len() {
            let was_hovered = self.targets[idx].hovered_audio;
            let now_hovered = self.targets[idx].item.kind == ContentType::Audio
                && self.targets[idx].contains(mouse_pos);
            let label = self.targets[idx].item.label.clone();

            if !was_hovered && now_hovered {
                self.play_item_audio(&label, sounds);
            } else if was_hovered && !now_hovered {
                self.stop_active_clip(sounds);
            }
            self.targets[idx].hovered_audio = now_hovered;
        }

        if is_mouse_button_pressed(MouseButton::Left) {
            if let Some(clicked_idx) = self.targets.iter().rposition(|t| t.contains(mouse_pos)) {
                let clicked = self.targets.remove(clicked_idx);
                if clicked.hovered_audio {
                    self.stop_active_clip(sounds);
                }
                let correct_id = self.deck[self.current_question.card_idx].id;

                if clicked.card_id == correct_id {
                    if self.wrong_answer_triggered {
                        self.advance_question(false, clicked.pos, Color::from_hex(0xFB7185), sounds);
                    } else {
                        if let Some(s) = &sounds.correct {
                            play_sound_once(s);
                        }
                        let used_hints = self.hints_revealed.iter().filter(|&&r| r).count();
                        let pts = match used_hints {
                            0 => 4,
                            1 => 3,
                            2 => 2,
                            _ => 1,
                        };
                        self.score += pts;
                        self.advance_question(true, clicked.pos, accent, sounds);
                    }
                } else {
                    if let Some(s) = &sounds.incorrect {
                        play_sound_once(s);
                    }
                    self.wrong_answer_triggered = true;
                    let correct_card_id = self.deck[self.current_question.card_idx].id;
                    self.targets.retain(|t| t.card_id == correct_card_id);
                    if self.targets.is_empty() {
                        self.spawn_target_for_card(self.current_question.card_idx);
                    }

                    if self.deck[self.current_question.card_idx].question.kind == ContentType::Audio
                    {
                        let label = self.deck[self.current_question.card_idx]
                            .question
                            .label
                            .clone();
                        self.play_item_audio(&label, sounds);
                        self.audio_replay_pulse = 0.6;
                    }
                }
            }
        }

        for pop in &mut self.pops {
            pop.timer += dt;
        }
        self.pops.retain(|p| p.timer < p.duration);

        false
    }
}

fn tile_grid_rect() -> Rect {
    let box_x = (screen_width() * 0.16).clamp(140.0, 320.0);
    let box_y = (screen_height() * 0.25).clamp(200.0, 300.0);
    let box_w = (screen_width() - box_x * 2.0).max(450.0);
    let box_h = (screen_height() - box_y - 40.0).max(350.0);
    let side = box_w.min(box_h);
    let origin_x = screen_width() / 2.0 - side / 2.0;
    let origin_y = box_y;
    Rect::new(origin_x, origin_y, side, side)
}

impl TileSession {
    async fn new(selected_entries: &[DeckEntry], difficulty: TileDifficulty, sounds: &SoundAssets) -> Self {
        let (deck, combined_deck_name) = load_combined_decks(selected_entries);

        let mut textures = HashMap::new();
        let mut audio_clips = HashMap::new();

        for card in &deck {
            let mut items = vec![&card.question, &card.answer];
            items.extend(card.extra_answers.iter());
            items.extend(card.hints.iter());

            for item in items {
                match item.kind {
                    ContentType::Image => {
                        if !textures.contains_key(&item.label) {
                            let resolved = resolve_deck_asset_path(None, &item.label)
                                .unwrap_or_else(|| item.label.clone());
                            if let Ok(tex) = load_texture(&resolved).await {
                                textures.insert(item.label.clone(), tex);
                            }
                        }
                    }
                    ContentType::Audio => {
                        if !audio_clips.contains_key(&item.label) {
                            let resolved = resolve_deck_asset_path(None, &item.label)
                                .unwrap_or_else(|| item.label.clone());
                            if let Ok(snd) = load_sound(&resolved).await {
                                audio_clips.insert(item.label.clone(), snd);
                            }
                        }
                    }
                    ContentType::Text => {}
                }
            }
        }

        let mut queue = VecDeque::new();
        let mut indices: Vec<usize> = (0..deck.len()).collect();
        shuffle_slice(&mut indices);
        for idx in indices {
            queue.push_back(QueuedQuestion {
                card_idx: idx,
                cohort_id: 0,
            });
        }

        let current_question = queue.pop_front().unwrap_or(QueuedQuestion {
            card_idx: 0,
            cohort_id: 0,
        });

        let q_time = difficulty.answer_time.max(0.5);
        let mut session = Self {
            deck_name: combined_deck_name,
            deck,
            difficulty: difficulty.clone(),
            textures,
            audio_clips,
            currently_playing_clip: None,
            queue,
            current_question,
            next_cohort_id: 1,
            loops_completed: 0,
            round_time_left: difficulty.round_time,
            question_time_left: q_time,
            score: 0,
            hints_revealed: [false; 3],
            audio_replay_pulse: 0.0,
            hint_audio_pulse: [0.0; 3],
            targets: Vec::new(),
            pops: Vec::new(),
            wrong_answer_triggered: false,
        };

        session.ensure_queue_refilled();
        session.populate_initial_targets();

        if !session.deck.is_empty()
            && session.deck[session.current_question.card_idx].question.kind == ContentType::Audio
        {
            let label = session.deck[session.current_question.card_idx]
                .question
                .label
                .clone();
            session.play_item_audio(&label, sounds);
            session.audio_replay_pulse = 0.6;
        }

        session
    }

    fn stop_active_clip(&mut self, sounds: &SoundAssets) {
        if let Some(prev_key) = self.currently_playing_clip.take() {
            if let Some(snd) = self.audio_clips.get(&prev_key) {
                stop_sound(snd);
            }
        }
        if let Some(s) = &sounds.placeholder_audio {
            stop_sound(s);
        }
    }

    fn play_item_audio(&mut self, label: &str, sounds: &SoundAssets) {
        self.stop_active_clip(sounds);
        if let Some(snd) = self.audio_clips.get(label) {
            play_sound_once(snd);
            self.currently_playing_clip = Some(label.to_string());
        } else if let Some(s) = &sounds.placeholder_audio {
            play_sound_once(s);
        }
    }

    fn ensure_queue_refilled(&mut self) {
        while self.queue.len() < 15 && !self.deck.is_empty() {
            let mut indices: Vec<usize> = (0..self.deck.len()).collect();
            shuffle_slice(&mut indices);
            let cohort = self.next_cohort_id;
            self.next_cohort_id += 1;
            for idx in indices {
                self.queue.push_back(QueuedQuestion {
                    card_idx: idx,
                    cohort_id: cohort,
                });
            }
        }
    }

    fn spawn_single_tile_for_card(&mut self, card_idx: usize) {
        let grid_size = self.difficulty.grid_size;
        let occupied: Vec<(usize, usize)> = self
            .targets
            .iter()
            .map(|t| (t.grid_row, t.grid_col))
            .collect();

        let mut free_cells = Vec::new();
        for r in 0..grid_size {
            for c in 0..grid_size {
                if !occupied.contains(&(r, c)) {
                    free_cells.push((r, c));
                }
            }
        }

        if let Some(&(r, c)) = free_cells.get(rand::gen_range(0, free_cells.len().max(1))) {
            let card = &self.deck[card_idx];
            self.targets.push(TileTarget {
                grid_row: r,
                grid_col: c,
                card_id: card.id,
                item: card.answer.clone(),
                spawn_scale: 0.01,
                hovered_audio: false,
            });
        }
    }

    fn populate_initial_targets(&mut self) {
        self.targets.clear();
        self.wrong_answer_triggered = false;

        let grid_size = self.difficulty.grid_size;
        let num_visible = self
            .difficulty
            .num_visible_tiles
            .clamp(3, grid_size * grid_size);

        self.ensure_queue_refilled();

        let mut queued_card_indices = vec![self.current_question.card_idx];
        for q in &self.queue {
            if !queued_card_indices.contains(&q.card_idx) {
                queued_card_indices.push(q.card_idx);
            }
        }
        shuffle_slice(&mut queued_card_indices);

        let mut chosen = vec![self.current_question.card_idx];
        for &idx in &queued_card_indices {
            if idx != self.current_question.card_idx && chosen.len() < num_visible {
                chosen.push(idx);
            }
        }

        while chosen.len() < num_visible {
            let rnd = rand::gen_range(0, self.deck.len());
            if !chosen.contains(&rnd) {
                chosen.push(rnd);
            }
        }

        shuffle_slice(&mut chosen);
        for &card_idx in &chosen {
            self.spawn_single_tile_for_card(card_idx);
        }
    }

    fn refill_tiles_for_new_question(&mut self) {
        if self.wrong_answer_triggered {
            return;
        }

        let grid_size = self.difficulty.grid_size;
        let desired = self
            .difficulty
            .num_visible_tiles
            .clamp(3, grid_size * grid_size);

        let correct_idx = self.current_question.card_idx;
        let correct_id = self.deck[correct_idx].id;

        if !self.targets.iter().any(|t| t.card_id == correct_id) {
            if self.targets.len() >= desired && !self.targets.is_empty() {
                let remove_idx = rand::gen_range(0, self.targets.len());
                self.targets.remove(remove_idx);
            }
            self.spawn_single_tile_for_card(correct_idx);
        }

        while self.targets.len() < desired {
            self.ensure_queue_refilled();
            let mut next_cand = self.current_question.card_idx;
            for q in &self.queue {
                if !self.targets.iter().any(|t| t.card_id == self.deck[q.card_idx].id) {
                    next_cand = q.card_idx;
                    break;
                }
            }
            self.spawn_single_tile_for_card(next_cand);
        }
    }

    fn advance_question(
        &mut self,
        answered_correctly: bool,
        pop_pos: Vec2,
        pop_color: Color,
        sounds: &SoundAssets,
    ) {
        self.pops.push(PopEffect {
            pos: pop_pos,
            base_radius: 56.0,
            timer: 0.0,
            duration: 0.35,
            color: pop_color,
        });

        let finished_q = self.current_question;
        if !answered_correctly {
            let insert_pos = self
                .queue
                .iter()
                .rposition(|q| q.cohort_id == finished_q.cohort_id)
                .map(|idx| idx + 1)
                .unwrap_or(0);
            self.queue.insert(insert_pos, finished_q);
        } else {
            let cohort_still_remaining = self
                .queue
                .iter()
                .any(|q| q.cohort_id == finished_q.cohort_id);
            if !cohort_still_remaining {
                self.loops_completed += 1;
            }
        }

        self.ensure_queue_refilled();
        if let Some(next_q) = self.queue.pop_front() {
            self.current_question = next_q;
        }

        self.question_time_left = self.difficulty.answer_time.max(0.5);
        self.hints_revealed = [false; 3];
        self.hint_audio_pulse = [0.0; 3];
        self.wrong_answer_triggered = false;

        self.refill_tiles_for_new_question();

        if self.deck[self.current_question.card_idx].question.kind == ContentType::Audio {
            let label = self.deck[self.current_question.card_idx]
                .question
                .label
                .clone();
            self.play_item_audio(&label, sounds);
            self.audio_replay_pulse = 0.6;
        }
    }

    fn update(
        &mut self,
        dt: f32,
        mouse_pos: Vec2,
        settings: &GameSettings,
        sounds: &SoundAssets,
    ) -> bool {
        let accent = Color::from_hex(settings.hex_accent);

        if self.audio_replay_pulse > 0.0 {
            self.audio_replay_pulse = (self.audio_replay_pulse - dt).max(0.0);
        }
        for p in &mut self.hint_audio_pulse {
            if *p > 0.0 {
                *p = (*p - dt).max(0.0);
            }
        }

        let replay_triggered = is_mouse_button_pressed(MouseButton::Right)
            || is_key_pressed(KeyCode::LeftShift)
            || is_key_pressed(KeyCode::RightShift);

        if replay_triggered
            && self.deck[self.current_question.card_idx].question.kind == ContentType::Audio
        {
            self.audio_replay_pulse = 0.6;
            let label = self.deck[self.current_question.card_idx]
                .question
                .label
                .clone();
            self.play_item_audio(&label, sounds);
        }

        if !self.difficulty.infinite_round {
            self.round_time_left -= dt;
            if self.round_time_left <= 0.0 {
                self.round_time_left = 0.0;
                return true;
            }
        }

        let total_q_time = self.difficulty.answer_time.max(0.5);
        if !self.wrong_answer_triggered {
            self.question_time_left -= dt;
        }
        let ratio_left = (self.question_time_left / total_q_time).clamp(0.0, 1.0);

        let num_hints = self.deck[self.current_question.card_idx].hints.len().min(3);
        let thresholds = [0.75, 0.50, 0.25];
        for i in 0..num_hints {
            let key_pressed = is_key_pressed(settings.hint_keys[i]);
            let auto_reveal =
                !self.wrong_answer_triggered && !self.hints_revealed[i] && ratio_left <= thresholds[i];

            if auto_reveal {
                self.hints_revealed[i] = true;
                if self.deck[self.current_question.card_idx].hints[i].kind == ContentType::Audio {
                    let label = self.deck[self.current_question.card_idx].hints[i]
                        .label
                        .clone();
                    self.play_item_audio(&label, sounds);
                    self.hint_audio_pulse[i] = 0.75;
                }
            } else if key_pressed {
                self.hints_revealed[i] = true;
                if self.deck[self.current_question.card_idx].hints[i].kind == ContentType::Audio {
                    let label = self.deck[self.current_question.card_idx].hints[i]
                        .label
                        .clone();
                    self.play_item_audio(&label, sounds);
                    self.hint_audio_pulse[i] = 0.75;
                }
            }
        }

        if is_key_pressed(KeyCode::Space) && !self.wrong_answer_triggered {
            if let Some(s) = &sounds.skipped {
                play_sound_once(s);
            }
            let q_pos = vec2(screen_width() / 2.0, 100.0);
            self.advance_question(false, q_pos, Color::from_hex(0xFB7185), sounds);
            return false;
        }

        if self.question_time_left <= 0.0 && !self.wrong_answer_triggered {
            if let Some(s) = &sounds.missed {
                play_sound_once(s);
            }
            let q_pos = vec2(screen_width() / 2.0, 100.0);
            self.advance_question(false, q_pos, Color::from_hex(0xFB7185), sounds);
            return false;
        }

        let grid_rect = tile_grid_rect();
        let cell_size = grid_rect.w / self.difficulty.grid_size as f32;

        for target in &mut self.targets {
            if target.spawn_scale < 1.0 {
                target.spawn_scale = (target.spawn_scale + dt * 4.5).min(1.0);
            }
        }

        for idx in 0..self.targets.len() {
            let was_hovered = self.targets[idx].hovered_audio;
            let now_hovered = self.targets[idx].item.kind == ContentType::Audio
                && self.targets[idx].contains(mouse_pos, grid_rect.point(), cell_size);
            let label = self.targets[idx].item.label.clone();

            if !was_hovered && now_hovered {
                self.play_item_audio(&label, sounds);
            } else if was_hovered && !now_hovered {
                self.stop_active_clip(sounds);
            }
            self.targets[idx].hovered_audio = now_hovered;
        }

        if is_mouse_button_pressed(MouseButton::Left) {
            if let Some(clicked_idx) = self
                .targets
                .iter()
                .rposition(|t| t.contains(mouse_pos, grid_rect.point(), cell_size))
            {
                let clicked = self.targets.remove(clicked_idx);
                if clicked.hovered_audio {
                    self.stop_active_clip(sounds);
                }
                let correct_id = self.deck[self.current_question.card_idx].id;
                let click_pos = grid_rect.point()
                    + vec2(
                        (clicked.grid_col as f32 + 0.5) * cell_size,
                        (clicked.grid_row as f32 + 0.5) * cell_size,
                    );

                if clicked.card_id == correct_id {
                    if self.wrong_answer_triggered {
                        self.advance_question(false, click_pos, Color::from_hex(0xFB7185), sounds);
                    } else {
                        if let Some(s) = &sounds.correct {
                            play_sound_once(s);
                        }
                        let used_hints = self.hints_revealed.iter().filter(|&&r| r).count();
                        let pts = match used_hints {
                            0 => 4,
                            1 => 3,
                            2 => 2,
                            _ => 1,
                        };
                        self.score += pts;
                        self.advance_question(true, click_pos, accent, sounds);
                    }
                } else {
                    if let Some(s) = &sounds.incorrect {
                        play_sound_once(s);
                    }
                    self.wrong_answer_triggered = true;
                    let correct_card_id = self.deck[self.current_question.card_idx].id;
                    self.targets.retain(|t| t.card_id == correct_card_id);
                    if self.targets.is_empty() {
                        self.spawn_single_tile_for_card(self.current_question.card_idx);
                    }

                    if self.deck[self.current_question.card_idx].question.kind == ContentType::Audio
                    {
                        let label = self.deck[self.current_question.card_idx]
                            .question
                            .label
                            .clone();
                        self.play_item_audio(&label, sounds);
                        self.audio_replay_pulse = 0.6;
                    }
                }
            }
        }

        for pop in &mut self.pops {
            pop.timer += dt;
        }
        self.pops.retain(|p| p.timer < p.duration);

        false
    }
}

impl DropSession {
    async fn new(selected_entries: &[DeckEntry], difficulty: DropDifficulty, sounds: &SoundAssets) -> Self {
        let (deck, combined_deck_name) = load_combined_decks(selected_entries);

        let mut textures = HashMap::new();
        let mut audio_clips = HashMap::new();

        for card in &deck {
            let mut items = vec![&card.question, &card.answer];
            items.extend(card.extra_answers.iter());
            items.extend(card.hints.iter());

            for item in items {
                match item.kind {
                    ContentType::Image => {
                        if !textures.contains_key(&item.label) {
                            let resolved = resolve_deck_asset_path(None, &item.label)
                                .unwrap_or_else(|| item.label.clone());
                            if let Ok(tex) = load_texture(&resolved).await {
                                textures.insert(item.label.clone(), tex);
                            }
                        }
                    }
                    ContentType::Audio => {
                        if !audio_clips.contains_key(&item.label) {
                            let resolved = resolve_deck_asset_path(None, &item.label)
                                .unwrap_or_else(|| item.label.clone());
                            if let Ok(snd) = load_sound(&resolved).await {
                                audio_clips.insert(item.label.clone(), snd);
                            }
                        }
                    }
                    ContentType::Text => {}
                }
            }
        }

        let mut queue = VecDeque::new();
        let mut indices: Vec<usize> = (0..deck.len()).collect();
        shuffle_slice(&mut indices);
        for idx in indices {
            queue.push_back(QueuedQuestion {
                card_idx: idx,
                cohort_id: 0,
            });
        }

        let current_question = queue.pop_front().unwrap_or(QueuedQuestion {
            card_idx: 0,
            cohort_id: 0,
        });

        let mut session = Self {
            deck_name: combined_deck_name,
            deck,
            difficulty: difficulty.clone(),
            textures,
            audio_clips,
            currently_playing_clip: None,
            queue,
            current_question,
            next_cohort_id: 1,
            loops_completed: 0,
            round_time_left: difficulty.round_time,
            score: 0,
            hints_revealed: [false; 3],
            audio_replay_pulse: 0.0,
            hint_audio_pulse: [0.0; 3],
            tiles: Vec::new(),
            pops: Vec::new(),
            wrong_answer_triggered: false,
        };

        session.ensure_queue_refilled();
        session.spawn_wave();

        if !session.deck.is_empty()
            && session.deck[session.current_question.card_idx].question.kind == ContentType::Audio
        {
            let label = session.deck[session.current_question.card_idx]
                .question
                .label
                .clone();
            session.play_item_audio(&label, sounds);
            session.audio_replay_pulse = 0.6;
        }

        session
    }

    fn stop_active_clip(&mut self, sounds: &SoundAssets) {
        if let Some(prev_key) = self.currently_playing_clip.take() {
            if let Some(snd) = self.audio_clips.get(&prev_key) {
                stop_sound(snd);
            }
        }
        if let Some(s) = &sounds.placeholder_audio {
            stop_sound(s);
        }
    }

    fn play_item_audio(&mut self, label: &str, sounds: &SoundAssets) {
        self.stop_active_clip(sounds);
        if let Some(snd) = self.audio_clips.get(label) {
            play_sound_once(snd);
            self.currently_playing_clip = Some(label.to_string());
        } else if let Some(s) = &sounds.placeholder_audio {
            play_sound_once(s);
        }
    }

    fn ensure_queue_refilled(&mut self) {
        while self.queue.len() < 15 && !self.deck.is_empty() {
            let mut indices: Vec<usize> = (0..self.deck.len()).collect();
            shuffle_slice(&mut indices);
            let cohort = self.next_cohort_id;
            self.next_cohort_id += 1;
            for idx in indices {
                self.queue.push_back(QueuedQuestion {
                    card_idx: idx,
                    cohort_id: cohort,
                });
            }
        }
    }

    fn spawn_wave(&mut self) {
        self.tiles.clear();
        self.wrong_answer_triggered = false;

        let num_lanes = self.difficulty.num_lanes.clamp(3, 7);
        let correct_idx = self.current_question.card_idx;

        let mut pool: Vec<usize> = (0..self.deck.len())
            .filter(|&i| i != correct_idx)
            .collect();
        shuffle_slice(&mut pool);

        let mut chosen_cards = vec![correct_idx];
        for i in 1..num_lanes {
            let pick = if !pool.is_empty() {
                pool[(i - 1) % pool.len()]
            } else {
                correct_idx
            };
            chosen_cards.push(pick);
        }
        shuffle_slice(&mut chosen_cards);

        let playfield = drop_playfield_rect();
        for (lane, &card_idx) in chosen_cards.iter().enumerate() {
            let card = &self.deck[card_idx];
            self.tiles.push(DropTile {
                lane,
                card_id: card.id,
                item: card.answer.clone(),
                y: playfield.y + 8.0,
                hovered_audio: false,
            });
        }
    }

    fn advance_question(
        &mut self,
        answered_correctly: bool,
        pop_pos: Vec2,
        pop_color: Color,
        sounds: &SoundAssets,
    ) {
        self.pops.push(PopEffect {
            pos: pop_pos,
            base_radius: 56.0,
            timer: 0.0,
            duration: 0.35,
            color: pop_color,
        });

        let finished_q = self.current_question;
        if !answered_correctly {
            let insert_pos = self
                .queue
                .iter()
                .rposition(|q| q.cohort_id == finished_q.cohort_id)
                .map(|idx| idx + 1)
                .unwrap_or(0);
            self.queue.insert(insert_pos, finished_q);
        } else {
            let cohort_still_remaining = self
                .queue
                .iter()
                .any(|q| q.cohort_id == finished_q.cohort_id);
            if !cohort_still_remaining {
                self.loops_completed += 1;
            }
        }

        self.ensure_queue_refilled();
        if let Some(next_q) = self.queue.pop_front() {
            self.current_question = next_q;
        }

        self.hints_revealed = [false; 3];
        self.hint_audio_pulse = [0.0; 3];
        self.spawn_wave();

        if self.deck[self.current_question.card_idx].question.kind == ContentType::Audio {
            let label = self.deck[self.current_question.card_idx]
                .question
                .label
                .clone();
            self.play_item_audio(&label, sounds);
            self.audio_replay_pulse = 0.6;
        }
    }

    fn update(
        &mut self,
        dt: f32,
        mouse_pos: Vec2,
        settings: &GameSettings,
        sounds: &SoundAssets,
    ) -> bool {
        let accent = Color::from_hex(settings.hex_accent);

        if self.audio_replay_pulse > 0.0 {
            self.audio_replay_pulse = (self.audio_replay_pulse - dt).max(0.0);
        }
        for p in &mut self.hint_audio_pulse {
            if *p > 0.0 {
                *p = (*p - dt).max(0.0);
            }
        }

        let replay_triggered = is_mouse_button_pressed(MouseButton::Right)
            || is_key_pressed(KeyCode::LeftShift)
            || is_key_pressed(KeyCode::RightShift);

        if replay_triggered
            && self.deck[self.current_question.card_idx].question.kind == ContentType::Audio
        {
            self.audio_replay_pulse = 0.6;
            let label = self.deck[self.current_question.card_idx]
                .question
                .label
                .clone();
            self.play_item_audio(&label, sounds);
        }

        if !self.difficulty.infinite_round {
            self.round_time_left -= dt;
            if self.round_time_left <= 0.0 {
                self.round_time_left = 0.0;
                return true;
            }
        }

        let playfield = drop_playfield_rect();
        let hit_line_y = playfield.y + playfield.h - 46.0;
        let travel_dist = (hit_line_y - (playfield.y + 8.0)).max(80.0);
        let current_y = self.tiles.first().map(|t| t.y).unwrap_or(playfield.y + 8.0);
        let ratio_left = (1.0 - ((current_y - (playfield.y + 8.0)) / travel_dist)).clamp(0.0, 1.0);

        let num_hints = self.deck[self.current_question.card_idx].hints.len().min(3);
        let thresholds = [0.75, 0.50, 0.25];
        for i in 0..num_hints {
            let key_pressed = is_key_pressed(settings.hint_keys[i]);
            let auto_reveal =
                !self.wrong_answer_triggered && !self.hints_revealed[i] && ratio_left <= thresholds[i];

            if auto_reveal || key_pressed {
                self.hints_revealed[i] = true;
                if self.deck[self.current_question.card_idx].hints[i].kind == ContentType::Audio {
                    let label = self.deck[self.current_question.card_idx].hints[i]
                        .label
                        .clone();
                    self.play_item_audio(&label, sounds);
                    self.hint_audio_pulse[i] = 0.75;
                }
            }
        }

        if is_key_pressed(KeyCode::Space) && !self.wrong_answer_triggered {
            if let Some(s) = &sounds.skipped {
                play_sound_once(s);
            }
            let q_pos = vec2(screen_width() / 2.0, 100.0);
            self.advance_question(false, q_pos, Color::from_hex(0xFB7185), sounds);
            return false;
        }

        let num_lanes = self.difficulty.num_lanes.clamp(3, 7);

        if !self.wrong_answer_triggered {
            for tile in &mut self.tiles {
                tile.y += self.difficulty.fall_speed * dt;
            }
        }

        if !self.wrong_answer_triggered {
            if let Some(first_tile) = self.tiles.first() {
                if first_tile.y + 78.0 >= hit_line_y {
                    if let Some(s) = &sounds.missed {
                        play_sound_once(s);
                    }
                    let q_pos = vec2(screen_width() / 2.0, 100.0);
                    self.advance_question(false, q_pos, Color::from_hex(0xFB7185), sounds);
                    return false;
                }
            }
        }

        for idx in 0..self.tiles.len() {
            let was_hovered = self.tiles[idx].hovered_audio;
            let now_hovered = self.tiles[idx].item.kind == ContentType::Audio
                && self.tiles[idx].contains(mouse_pos, playfield, num_lanes);
            let label = self.tiles[idx].item.label.clone();

            if !was_hovered && now_hovered {
                self.play_item_audio(&label, sounds);
            } else if was_hovered && !now_hovered {
                self.stop_active_clip(sounds);
            }
            self.tiles[idx].hovered_audio = now_hovered;
        }

        if is_mouse_button_pressed(MouseButton::Left) {
            let correct_id = self.deck[self.current_question.card_idx].id;

            if self.wrong_answer_triggered {
                if let Some(correct_tile) = self
                    .tiles
                    .iter()
                    .find(|t| t.card_id == correct_id && t.contains(mouse_pos, playfield, num_lanes))
                {
                    let pos = correct_tile.rect(playfield, num_lanes).center();
                    self.advance_question(false, pos, Color::from_hex(0xFB7185), sounds);
                }
            } else if let Some(clicked_idx) = self
                .tiles
                .iter()
                .rposition(|t| t.contains(mouse_pos, playfield, num_lanes))
            {
                let clicked = self.tiles[clicked_idx].clone();
                if clicked.hovered_audio {
                    self.stop_active_clip(sounds);
                }
                let click_pos = clicked.rect(playfield, num_lanes).center();

                if clicked.card_id == correct_id {
                    if let Some(s) = &sounds.correct {
                        play_sound_once(s);
                    }
                    let used_hints = self.hints_revealed.iter().filter(|&&r| r).count();
                    let pts = match used_hints {
                        0 => 4,
                        1 => 3,
                        2 => 2,
                        _ => 1,
                    };
                    self.score += pts;
                    self.advance_question(true, click_pos, accent, sounds);
                } else {
                    if let Some(s) = &sounds.incorrect {
                        play_sound_once(s);
                    }
                    self.wrong_answer_triggered = true;
                    self.tiles.retain(|t| t.card_id == correct_id);
                    if self.tiles.is_empty() {
                        let card = &self.deck[self.current_question.card_idx];
                        self.tiles.push(DropTile {
                            lane: num_lanes / 2,
                            card_id: card.id,
                            item: card.answer.clone(),
                            y: clicked.y,
                            hovered_audio: false,
                        });
                    }
                    if self.deck[self.current_question.card_idx].question.kind == ContentType::Audio
                    {
                        let label = self.deck[self.current_question.card_idx]
                            .question
                            .label
                            .clone();
                        self.play_item_audio(&label, sounds);
                        self.audio_replay_pulse = 0.6;
                    }
                }
            }
        }

        for pop in &mut self.pops {
            pop.timer += dt;
        }
        self.pops.retain(|p| p.timer < p.duration);

        false
    }
}

// CUSTOM CURSOR IMPLEMENTATION
// ============================================================================
struct CustomCursor {
    history: VecDeque<Vec2>,
    prev_pos: Vec2,
    smoothed_speed: f32,
    displayed_length: f32,
}

impl CustomCursor {
    fn new() -> Self {
        let start_pos: Vec2 = mouse_position().into();
        Self {
            history: VecDeque::with_capacity(MAX_TRAIL_LENGTH_CAP as usize),
            prev_pos: start_pos,
            smoothed_speed: 0.0,
            displayed_length: 0.0,
        }
    }

    fn update_and_draw(&mut self, dt: f32, settings: &GameSettings) {
        let dt = dt.max(0.0001);
        let mouse_pos: Vec2 = mouse_position().into();

        let in_game_canvas = mouse_pos.x >= 0.0
            && mouse_pos.x <= screen_width()
            && mouse_pos.y >= 2.0
            && mouse_pos.y <= screen_height();

        if !in_game_canvas {
            show_mouse(true);
            self.history.clear();
            self.displayed_length = 0.0;
            self.smoothed_speed = 0.0;
            self.prev_pos = mouse_pos;
            return;
        }

        show_mouse(false);

        let dot_radius = settings.dot_radius;
        let trail_max_radius = dot_radius * settings.trail_radius_mult;
        let idle_pulse_max_radius = dot_radius * IDLE_PULSE_MAX_MULT;
        let max_trail_len = settings.max_trail_length;
        let max_segments =
            (max_trail_len.round() as usize).clamp(2, MAX_TRAIL_LENGTH_CAP as usize);

        let dot_color = Color::from_hex(settings.hex_cursor_dot);
        let base_trail_color = Color::from_hex(settings.hex_cursor_trail);

        let instant_speed = (mouse_pos - self.prev_pos).length() / dt;
        self.smoothed_speed = self.smoothed_speed * 0.8 + instant_speed * 0.2;
        self.prev_pos = mouse_pos;

        let velocity_ratio = (self.smoothed_speed / MAX_CURSOR_SPEED).clamp(0.0, 1.0);

        let target_length = max_trail_len * velocity_ratio;
        let lerp_rate = if target_length > self.displayed_length {
            LENGTH_GROW_SPEED
        } else {
            LENGTH_SHRINK_SPEED
        };
        let blend = (lerp_rate * dt).clamp(0.0, 1.0);
        self.displayed_length += (target_length - self.displayed_length) * blend;

        self.history.push_front(mouse_pos);
        while self.history.len() > max_segments {
            self.history.pop_back();
        }

        if self.displayed_length > 1.0 && self.history.len() >= 2 {
            let mut active_points = vec![self.history[0]];
            let mut accumulated_dist = 0.0;

            for i in 1..self.history.len() {
                let seg_len = (self.history[i] - self.history[i - 1]).length();
                if accumulated_dist + seg_len > self.displayed_length {
                    let remaining = self.displayed_length - accumulated_dist;
                    if seg_len > 0.001 {
                        let dir = (self.history[i] - self.history[i - 1]) / seg_len;
                        active_points.push(self.history[i - 1] + dir * remaining);
                    }
                    accumulated_dist = self.displayed_length;
                    break;
                }
                accumulated_dist += seg_len;
                active_points.push(self.history[i]);
            }

            if active_points.len() >= 2 && accumulated_dist > 1.0 {
                let mut cum_dists = vec![0.0];
                let mut dist_so_far = 0.0;
                for i in 1..active_points.len() {
                    dist_so_far += (active_points[i] - active_points[i - 1]).length();
                    cum_dists.push(dist_so_far);
                }

                let tip_alpha = velocity_ratio.powf(1.6);

                for i in (1..active_points.len()).rev() {
                    let p0 = active_points[i - 1];
                    let p1 = active_points[i];
                    let d0 = cum_dists[i - 1];
                    let d1 = cum_dists[i];
                    let seg_len = (p1 - p0).length();

                    let steps = (seg_len / 1.5).ceil().max(1.0) as usize;
                    for s in (0..=steps).rev() {
                        let frac = s as f32 / steps as f32;
                        let pos = p0.lerp(p1, frac);
                        let dist_from_head = d0 + (d1 - d0) * frac;
                        let progress = (dist_from_head / accumulated_dist).clamp(0.0, 1.0);

                        let radius = trail_max_radius * (1.0 - progress);
                        let alpha = 1.0 + (tip_alpha - 1.0) * progress.powf(0.85);

                        if radius > 0.15 && alpha > 0.01 {
                            let mut seg_color = base_trail_color;
                            seg_color.a = alpha;
                            draw_circle(pos.x, pos.y, radius, seg_color);
                        }
                    }
                }
            }
        }

        let idle_factor = (1.0 - velocity_ratio * 2.5).clamp(0.0, 1.0);
        if idle_factor > 0.01 {
            let wave = ((get_time() as f32 * 3.2).sin() + 1.0) * 0.5;
            let pulse_radius = dot_radius + (idle_pulse_max_radius - dot_radius) * wave;

            let mut pulse_color = base_trail_color;
            pulse_color.a = 0.45 * idle_factor;
            draw_circle(mouse_pos.x, mouse_pos.y, pulse_radius, pulse_color);
        }

        draw_circle(mouse_pos.x, mouse_pos.y, dot_radius, dot_color);
    }
}

fn scan_available_decks() -> Vec<DeckEntry> {
    let _ = fs::create_dir_all(DECKS_DIR);
    let mut decks = Vec::new();

    if let Ok(entries) = fs::read_dir(DECKS_DIR) {
        let mut discovered = Vec::new();
        for entry in entries.flatten() {
            let path = entry.path();
            if path.is_dir() {
                if let Ok(sub_entries) = fs::read_dir(&path) {
                    let csv_in_dir = sub_entries.flatten().map(|e| e.path()).find(|p| {
                        p.extension()
                            .and_then(|ext| ext.to_str())
                            .map(|ext| {
                                matches!(ext.to_ascii_lowercase().as_str(), "csv" | "fmc" | "txt")
                            })
                            .unwrap_or(false)
                    });
                    if let Some(csv_file) = csv_in_dir {
                        let folder_name = path
                            .file_name()
                            .and_then(|n| n.to_str())
                            .unwrap_or("Deck Folder")
                            .to_string();
                        discovered.push(DeckEntry {
                            name: folder_name,
                            csv_path: Some(csv_file),
                            base_dir: Some(path),
                        });
                    }
                }
            } else if path.is_file() {
                let is_deck = path
                    .extension()
                    .and_then(|ext| ext.to_str())
                    .map(|ext| matches!(ext.to_ascii_lowercase().as_str(), "csv" | "fmc" | "txt"))
                    .unwrap_or(false);
                if is_deck {
                    let name = path
                        .file_name()
                        .and_then(|n| n.to_str())
                        .unwrap_or("Deck.csv")
                        .to_string();
                    let parent = path.parent().map(|p| p.to_path_buf());
                    discovered.push(DeckEntry {
                        name,
                        csv_path: Some(path),
                        base_dir: parent,
                    });
                }
            }
        }
        discovered.sort_by(|a, b| a.name.cmp(&b.name));
        decks.extend(discovered);
    }

    decks
}

fn copy_dir_recursive(src: &Path, dst: &Path) -> Result<(), String> {
    fs::create_dir_all(dst)
        .map_err(|e| format!("Failed to create folder {}: {}", dst.display(), e))?;
    let entries = fs::read_dir(src)
        .map_err(|e| format!("Failed to read folder {}: {}", src.display(), e))?;
    for entry in entries.flatten() {
        let entry_path = entry.path();
        let target_path = dst.join(entry.file_name());
        if entry_path.is_dir() {
            copy_dir_recursive(&entry_path, &target_path)?;
        } else if entry_path.is_file() {
            fs::copy(&entry_path, &target_path)
                .map_err(|e| format!("Failed to copy {}: {}", entry_path.display(), e))?;
        }
    }
    Ok(())
}

fn import_deck_path(src_path: &Path) -> Result<String, String> {
    let _ = fs::create_dir_all(DECKS_DIR);
    
    let cleaned_path = if src_path.to_string_lossy().ends_with(['/', '\\']) {
        let s = src_path.to_string_lossy();
        PathBuf::from(&s[..s.len() - 1])
    } else {
        src_path.to_path_buf()
    };

    let item_name = cleaned_path
        .file_name()
        .and_then(|n| n.to_str())
        .ok_or_else(|| "Invalid file or folder name".to_string())?;

    if cleaned_path.is_dir() {
        let dest_dir = Path::new(DECKS_DIR).join(item_name);
        copy_dir_recursive(&cleaned_path, &dest_dir)?;
        Ok(item_name.to_string())
    } else if cleaned_path.is_file() {
        let dest_file = Path::new(DECKS_DIR).join(item_name);
        fs::copy(&cleaned_path, &dest_file).map_err(|e| format!("Failed to copy deck: {}", e))?;
        Ok(item_name.to_string())
    } else {
        Err("Path not found or cannot be accessed.".to_string())
    }
}

fn spawn_folder_picker() -> Receiver<String> {
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        #[cfg(target_os = "windows")]
        {
            let ps_script = "Add-Type -AssemblyName System.Windows.Forms; \
                $f = New-Object System.Windows.Forms.FolderBrowserDialog; \
                $f.Description = 'Select your FlashMyCards Deck Folder (containing the .csv and media subfolders)'; \
                if ($f.ShowDialog() -eq 'OK') { Write-Output $f.SelectedPath }";
            if let Ok(output) = std::process::Command::new("powershell")
                .args(["-NoProfile", "-Command", ps_script])
                .output()
            {
                let selected = String::from_utf8_lossy(&output.stdout).trim().to_string();
                if !selected.is_empty() {
                    let _ = tx.send(selected);
                }
            }
        }
    });
    rx
}

fn rounded_rect_perimeter(x: f32, y: f32, w: f32, h: f32, radius: f32) -> Vec<Vec2> {
    let r = radius.min(w * 0.48).min(h * 0.48).max(1.0);
    let corner_steps = 10;
    let mut pts = Vec::with_capacity(corner_steps * 4 + 4);

    let corners = [
        (vec2(x + w - r, y + r), -std::f32::consts::FRAC_PI_2, 0.0),
        (vec2(x + w - r, y + h - r), 0.0, std::f32::consts::FRAC_PI_2),
        (vec2(x + r, y + h - r), std::f32::consts::FRAC_PI_2, std::f32::consts::PI),
        (vec2(x + r, y + r), std::f32::consts::PI, 3.0 * std::f32::consts::FRAC_PI_2),
    ];

    for (center, start_ang, end_ang) in corners {
        for s in 0..=corner_steps {
            let t = s as f32 / corner_steps as f32;
            let ang = start_ang + (end_ang - start_ang) * t;
            pts.push(vec2(center.x + ang.cos() * r, center.y + ang.sin() * r));
        }
    }

    pts
}

fn draw_rounded_box(x: f32, y: f32, w: f32, h: f32, r: f32, fill: Color, border: Color) {
    let border_thickness = 2.0;
    let outer = rounded_rect_perimeter(x, y, w, h, r);
    let inner = rounded_rect_perimeter(
        x + border_thickness,
        y + border_thickness,
        (w - border_thickness * 2.0).max(2.0),
        (h - border_thickness * 2.0).max(2.0),
        (r - border_thickness).max(1.0),
    );

    let center = vec2(x + w * 0.5, y + h * 0.5);
    let n = inner.len();
    for i in 0..n {
        let next = (i + 1) % n;
        draw_triangle(center, inner[i], inner[next], fill);
    }

    for i in 0..n {
        let next = (i + 1) % n;
        draw_triangle(outer[i], outer[next], inner[next], border);
        draw_triangle(outer[i], inner[next], inner[i], border);
    }
}

fn draw_button(
    label: &str,
    center_x: f32,
    y: f32,
    font_size: u16,
    mouse_pos: Vec2,
    text_primary: Color,
    text_muted: Color,
    accent: Color,
    font: Option<&Font>,
) -> bool {
    let dims = measure_text(label, font, font_size, 1.0);
    let x = center_x - dims.width / 2.0;
    let pad_x = 22.0;
    let pad_y = 12.0;
    let rect = Rect::new(
        x - pad_x,
        y - dims.height - pad_y / 2.0,
        dims.width + pad_x * 2.0,
        dims.height + pad_y,
    );

    let hovered = rect.contains(mouse_pos);
    if hovered {
        let mut fill = accent;
        fill.a = 0.18;
        let mut border = accent;
        border.a = 0.85;
        draw_rectangle(rect.x, rect.y, rect.w, rect.h, fill);
        draw_rectangle_lines(rect.x, rect.y, rect.w, rect.h, 2.0, border);
    }

    let color = if hovered { text_primary } else { text_muted };
    draw_ui_text(label, x, y, font_size, color, font);

    hovered && is_mouse_button_pressed(MouseButton::Left)
}

fn draw_slider(
    label: &str,
    value_text: &str,
    value: f32,
    min: f32,
    max: f32,
    x: f32,
    y: f32,
    width: f32,
    mouse_pos: Vec2,
    settings: &GameSettings,
    font: Option<&Font>,
) -> f32 {
    let text_primary = Color::from_hex(settings.hex_text_primary);
    let text_muted = Color::from_hex(settings.hex_text_muted);
    let accent = Color::from_hex(settings.hex_accent);

    draw_ui_text(label, x, y, 17, text_primary, font);
    let val_dims = measure_text(value_text, font, 17, 1.0);
    draw_ui_text(value_text, x + width - val_dims.width, y, 17, accent, font);

    let track_y = y + 9.0;
    let track_h = 6.0;
    let hit_rect = Rect::new(x - 8.0, track_y - 9.0, width + 16.0, track_h + 18.0);

    let mut new_val = value;
    if hit_rect.contains(mouse_pos) && is_mouse_button_down(MouseButton::Left) {
        let ratio = ((mouse_pos.x - x) / width).clamp(0.0, 1.0);
        new_val = min + ratio * (max - min);
    }

    let ratio = ((new_val - min) / (max - min)).clamp(0.0, 1.0);
    let fill_w = width * ratio;

    let mut track_bg = text_muted;
    track_bg.a = 0.25;
    draw_rectangle(x, track_y, width, track_h, track_bg);
    draw_rectangle(x, track_y, fill_w, track_h, accent);
    draw_circle(x + fill_w, track_y + track_h / 2.0, 6.5, text_primary);

    new_val
}

#[cfg(target_os = "windows")]
fn disable_window_drag_drop(hwnd: isize) {
    #[link(name = "ole32")]
    unsafe extern "system" {
        fn RevokeDragDrop(hwnd: isize) -> i32;
    }
    #[link(name = "shell32")]
    unsafe extern "system" {
        fn DragAcceptFiles(hwnd: isize, fAccept: i32);
    }
    if hwnd != 0 {
        unsafe {
            let _ = RevokeDragDrop(hwnd);
            DragAcceptFiles(hwnd, 0);
        }
    }
}

#[cfg(target_os = "windows")]
fn apply_windows_ico(ico_path: &str) {
    use std::ffi::OsStr;
    use std::os::windows::ffi::OsStrExt;

    unsafe extern "system" {
        fn GetActiveWindow() -> isize;
        fn GetForegroundWindow() -> isize;
        fn LoadImageW(
            hInst: isize,
            name: *const u16,
            type_: u32,
            cx: i32,
            cy: i32,
            fuLoad: u32,
        ) -> isize;
        fn SendMessageW(hwnd: isize, msg: u32, wparam: usize, lparam: isize) -> isize;
        fn SetClassLongPtrW(hwnd: isize, nIndex: i32, dwNewLong: isize) -> isize;
    }

    const IMAGE_ICON: u32 = 1;
    const LR_LOADFROMFILE: u32 = 0x00000010;
    const WM_SETICON: u32 = 0x0080;
    const ICON_SMALL: usize = 0;
    const ICON_BIG: usize = 1;
    const GCLP_HICON: i32 = -14;
    const GCLP_HICONSM: i32 = -34;

    unsafe {
        let mut hwnd = GetActiveWindow();
        if hwnd == 0 {
            hwnd = GetForegroundWindow();
        }
        disable_window_drag_drop(hwnd);
    }

    if !Path::new(ico_path).is_file() {
        return;
    }

    let wide_path: Vec<u16> = OsStr::new(ico_path)
        .encode_wide()
        .chain(std::iter::once(0))
        .collect();

    unsafe {
        let mut hwnd = GetActiveWindow();
        if hwnd == 0 {
            hwnd = GetForegroundWindow();
        }
        if hwnd != 0 {
            disable_window_drag_drop(hwnd);
            let hicon_big = LoadImageW(
                0,
                wide_path.as_ptr(),
                IMAGE_ICON,
                48,
                48,
                LR_LOADFROMFILE,
            );
            let hicon_small = LoadImageW(
                0,
                wide_path.as_ptr(),
                IMAGE_ICON,
                16,
                16,
                LR_LOADFROMFILE,
            );
            if hicon_big != 0 {
                SendMessageW(hwnd, WM_SETICON, ICON_BIG, hicon_big);
                SetClassLongPtrW(hwnd, GCLP_HICON, hicon_big);
            }
            if hicon_small != 0 {
                SendMessageW(hwnd, WM_SETICON, ICON_SMALL, hicon_small);
                SetClassLongPtrW(hwnd, GCLP_HICONSM, hicon_small);
            }
        }
    }
}

#[cfg(not(target_os = "windows"))]
fn apply_windows_ico(_ico_path: &str) {}

fn window_conf() -> Conf {
    let (w, h) = detect_monitor_resolution();
    Conf {
        window_title: "FlashMyCards".to_owned(),
        window_width: w,
        window_height: h,
        window_resizable: true,
        fullscreen: false,
        ..Default::default()
    }
}

#[macroquad::main(window_conf)]
async fn main() {
    apply_windows_ico("assets/FMC.ico");
    let mut icon_applied_frames = 0u8;

    rand::srand(miniquad::date::now() as u64);
    let startup_subtitle = STARTUP_SUBTITLES[rand::gen_range(0, STARTUP_SUBTITLES.len())];

    let custom_font = load_ttf_font("assets/NotoSansJP-VariableFont_wght.ttf").await.ok();

    let sounds = SoundAssets {
        correct: load_sound("assets/CorrectAnswer.wav").await.ok(),
        incorrect: load_sound("assets/IncorrectAnswer.wav").await.ok(),
        missed: load_sound("assets/MissedAnswer.wav").await.ok(),
        skipped: load_sound("assets/SkippedAnswer.wav").await.ok(),
        placeholder_audio: load_sound("assets/Bababoey.wav").await.ok(),
    };

    let mut settings = GameSettings::default();
    let mut current_state = GameState::Menu;
    let mut selected_index = 0;
    let mut custom_cursor = CustomCursor::new();

    let mut available_decks = scan_available_decks();
    let mut selected_deck_index: usize = 0;
    let mut selected_deck_names: Vec<String> = available_decks
        .first()
        .map(|d| vec![d.name.clone()])
        .unwrap_or_default();
    let mut deck_status_msg = String::new();
    let mut folder_picker_rx: Option<Receiver<String>> = None;

    let mut active_hex_slot: Option<ColorSlot> = None;
    let mut hex_input_buffer = String::new();
    let mut rebinding_hint_idx: Option<usize> = None;

    let mut float_difficulty = FloatDifficulty::default();
    let mut tile_difficulty = TileDifficulty::default();
    let mut drop_difficulty = DropDifficulty::default();

    let mut float_session: Option<FloatSession> = None;
    let mut tile_session: Option<TileSession> = None;
    let mut drop_session: Option<DropSession> = None;
    let mut deck_high_scores: HashMap<String, u32> = HashMap::new();

    let mut mode_deck_scroll_offset: f32 = 0.0;

    let preview_bubbles_init = [
        vec2(0.5, 0.5),
        vec2(0.25, 0.25),
        vec2(0.75, 0.25),
        vec2(0.25, 0.75),
        vec2(0.75, 0.75),
    ];

    struct TileIconCell {
        row: usize,
        col: usize,
        scale: f32,
    }
    let mut tile_icon_cells = vec![
        TileIconCell { row: 0, col: 1, scale: 1.0 },
        TileIconCell { row: 1, col: 2, scale: 1.0 },
        TileIconCell { row: 2, col: 0, scale: 1.0 },
        TileIconCell { row: 3, col: 3, scale: 1.0 },
    ];
    let tile_loop_pattern: &[(usize, usize)] = &[
        (0, 3),
        (2, 2),
        (1, 0),
        (3, 1),
        (0, 1),
        (1, 2),
        (2, 0),
        (3, 3),
    ];
    let mut tile_pattern_idx: usize = 0;
    let mut tile_anim_timer: f32 = 0.0;
    let mut tile_icon_pops: Vec<PopEffect> = Vec::new();

    let mut drop_icon_y_offset: f32 = 0.0;
    let mut drop_icon_pause_timer: f32 = 0.0;

    loop {
        if icon_applied_frames < 5 {
            apply_windows_ico("assets/FMC.ico");
            icon_applied_frames += 1;
        }

        let is_penalty_blackout = match current_state {
            GameState::FloatPlaying => {
                if let Some(s) = &float_session {
                    s.wrong_answer_triggered
                } else {
                    false
                }
            }
            GameState::TilePlaying => {
                if let Some(s) = &tile_session {
                    s.wrong_answer_triggered
                } else {
                    false
                }
            }
            GameState::DropPlaying => {
                if let Some(s) = &drop_session {
                    s.wrong_answer_triggered
                } else {
                    false
                }
            }
            _ => false,
        };

        let bg_color = if is_penalty_blackout {
            Color::new(0.0, 0.0, 0.0, 1.0)
        } else {
            Color::from_hex(settings.hex_bg)
        };
        clear_background(bg_color);

        let dt = get_frame_time();
        let mouse_pos: Vec2 = mouse_position().into();

        let text_primary = Color::from_hex(settings.hex_text_primary);
        let text_muted = Color::from_hex(settings.hex_text_muted);
        let accent_color = Color::from_hex(settings.hex_accent);

        if let Some(rx) = &folder_picker_rx {
            if let Ok(picked_path) = rx.try_recv() {
                match import_deck_path(Path::new(&picked_path)) {
                    Ok(name) => {
                        deck_status_msg = format!("Imported deck folder: {}", name);
                        available_decks = scan_available_decks();
                        if let Some(pos) = available_decks.iter().position(|d| d.name == name) {
                            selected_deck_index = pos;
                        }
                        if !selected_deck_names.contains(&name) {
                            selected_deck_names.push(name);
                        }
                    }
                    Err(err) => deck_status_msg = err,
                }
                folder_picker_rx = None;
            }
        }

        // Keep selected_deck_names synced with available_decks (ensure at least 1 selected if decks exist)
        selected_deck_names.retain(|n| available_decks.iter().any(|d| &d.name == n));
        if selected_deck_names.is_empty() && !available_decks.is_empty() {
            let fallback_idx = selected_deck_index.min(available_decks.len() - 1);
            selected_deck_names.push(available_decks[fallback_idx].name.clone());
        }

        // ====================================================================
        // SCREEN RENDERING
        // ====================================================================
        match current_state {
            GameState::Menu => {
                draw_centered_text(
                    "FLASH MY CARDS",
                    150.0,
                    52,
                    text_primary,
                    custom_font.as_ref(),
                );
                draw_centered_text(startup_subtitle, 195.0, 20, text_muted, custom_font.as_ref());

                let start_y = 290.0;
                let spacing = 60.0;

                for (i, (label, option)) in MENU_OPTIONS.iter().enumerate() {
                    let font_size = 28;
                    let y = start_y + (i as f32 * spacing);

                    let dims = measure_text(label, custom_font.as_ref(), font_size, 1.0);
                    let x = screen_width() / 2.0 - dims.width / 2.0;

                    let padding_x = 24.0;
                    let padding_y = 14.0;
                    let rect = Rect::new(
                        x - padding_x,
                        y - dims.height - padding_y / 2.0,
                        dims.width + padding_x * 2.0,
                        dims.height + padding_y,
                    );

                    let is_hovered = rect.contains(mouse_pos);
                    if is_hovered {
                        selected_index = i;
                    }

                    let is_selected = selected_index == i;

                    if is_selected {
                        let mut fill = accent_color;
                        fill.a = 0.18;
                        let mut border = accent_color;
                        border.a = 0.85;

                        draw_rectangle(rect.x, rect.y, rect.w, rect.h, fill);
                        draw_rectangle_lines(rect.x, rect.y, rect.w, rect.h, 2.0, border);
                    }

                    let label_color = if is_selected {
                        text_primary
                    } else {
                        text_muted
                    };
                    draw_ui_text(label, x, y, font_size, label_color, custom_font.as_ref());

                    if is_hovered && is_mouse_button_pressed(MouseButton::Left) {
                        match option {
                            MenuOption::Play => {
                                available_decks = scan_available_decks();
                                if available_decks.is_empty() {
                                    deck_status_msg =
                                        "You have loaded no decks. Please import a deck."
                                            .to_string();
                                    current_state = GameState::EditDeck;
                                } else {
                                    current_state = GameState::ModeSelect;
                                }
                            }
                            MenuOption::EditDeck => {
                                available_decks = scan_available_decks();
                                current_state = GameState::EditDeck;
                            }
                            MenuOption::Settings => {
                                active_hex_slot = None;
                                rebinding_hint_idx = None;
                                current_state = GameState::Settings;
                            }
                            MenuOption::Quit => std::process::exit(0),
                        }
                    }
                }

                if is_key_pressed(KeyCode::Down) || is_key_pressed(KeyCode::S) {
                    selected_index = (selected_index + 1) % MENU_OPTIONS.len();
                }
                if is_key_pressed(KeyCode::Up) || is_key_pressed(KeyCode::W) {
                    selected_index = (selected_index + MENU_OPTIONS.len() - 1) % MENU_OPTIONS.len();
                }
                if is_key_pressed(KeyCode::Enter) || is_key_pressed(KeyCode::Space) {
                    match MENU_OPTIONS[selected_index].1 {
                        MenuOption::Play => {
                            available_decks = scan_available_decks();
                            if available_decks.is_empty() {
                                deck_status_msg =
                                    "You have loaded no decks. Please import a deck.".to_string();
                                current_state = GameState::EditDeck;
                            } else {
                                current_state = GameState::ModeSelect;
                            }
                        }
                        MenuOption::EditDeck => {
                            available_decks = scan_available_decks();
                            current_state = GameState::EditDeck;
                        }
                        MenuOption::Settings => {
                            active_hex_slot = None;
                            rebinding_hint_idx = None;
                            current_state = GameState::Settings;
                        }
                        MenuOption::Quit => std::process::exit(0),
                    }
                }
            }

            GameState::EditDeck => {
                draw_centered_text("ADD DECK", 70.0, 44, text_primary, custom_font.as_ref());

                let panel_w = 540.0_f32.min(screen_width() - 60.0);
                let panel_x = screen_width() / 2.0 - panel_w / 2.0;

                draw_ui_text(
                    "AVAILABLE DECKS (assets/decks)",
                    panel_x,
                    130.0,
                    18,
                    text_muted,
                    custom_font.as_ref(),
                );

                let list_top = 145.0;
                let list_h = 180.0;
                let mut list_bg = text_muted;
                list_bg.a = 0.08;
                draw_rectangle(panel_x, list_top, panel_w, list_h, list_bg);
                draw_rectangle_lines(panel_x, list_top, panel_w, list_h, 1.5, text_muted);

                if available_decks.is_empty() {
                    draw_centered_text(
                        "You have loaded no decks. Please import a deck.",
                        list_top + list_h / 2.0 - 10.0,
                        18,
                        accent_color,
                        custom_font.as_ref(),
                    );
                } else {
                    for (i, deck_entry) in available_decks.iter().enumerate().take(5) {
                        let item_y = list_top + 32.0 + (i as f32 * 32.0);
                        let item_rect =
                            Rect::new(panel_x + 8.0, item_y - 20.0, panel_w - 16.0, 28.0);
                        let hovered = item_rect.contains(mouse_pos);
                        let is_sel = selected_deck_index == i;

                        if is_sel || hovered {
                            let mut hl = accent_color;
                            hl.a = if is_sel { 0.22 } else { 0.12 };
                            draw_rectangle(item_rect.x, item_rect.y, item_rect.w, item_rect.h, hl);
                        }

                        if hovered && is_mouse_button_pressed(MouseButton::Left) {
                            selected_deck_index = i;
                        }

                        let row_label = format!("{}. {}", i + 1, deck_entry.name);
                        draw_ui_text(
                            &row_label,
                            panel_x + 18.0,
                            item_y,
                            18,
                            if is_sel { text_primary } else { text_muted },
                            custom_font.as_ref(),
                        );
                    }
                }

                let drop_top = list_top + list_h + 20.0;
                let drop_h = 100.0;
                let drop_rect = Rect::new(panel_x, drop_top, panel_w, drop_h);
                let drop_hovered = drop_rect.contains(mouse_pos);

                let mut drop_bg = accent_color;
                drop_bg.a = if drop_hovered { 0.18 } else { 0.08 };
                draw_rectangle(drop_rect.x, drop_rect.y, drop_rect.w, drop_rect.h, drop_bg);
                draw_rectangle_lines(
                    drop_rect.x,
                    drop_rect.y,
                    drop_rect.w,
                    drop_rect.h,
                    2.0,
                    if drop_hovered {
                        text_primary
                    } else {
                        accent_color
                    },
                );

                draw_centered_text(
                    "CLICK TO SELECT DECK FOLDER",
                    drop_top + 38.0,
                    20,
                    text_primary,
                    custom_font.as_ref(),
                );
                draw_centered_text(
                    "select folder containing .csv and media subfolders",
                    drop_top + 66.0,
                    14,
                    text_muted,
                    custom_font.as_ref(),
                );

                if drop_hovered && is_mouse_button_pressed(MouseButton::Left) {
                    folder_picker_rx = Some(spawn_folder_picker());
                }

                if !deck_status_msg.is_empty() {
                    draw_centered_text(
                        &deck_status_msg,
                        drop_top + drop_h + 16.0,
                        16,
                        accent_color,
                        custom_font.as_ref(),
                    );
                }

                let bottom_y = screen_height() - 40.0;
                if draw_button(
                    "SELECT MODE",
                    screen_width() / 2.0,
                    bottom_y - 52.0,
                    22,
                    mouse_pos,
                    text_primary,
                    text_muted,
                    accent_color,
                    custom_font.as_ref(),
                ) {
                    available_decks = scan_available_decks();
                    if available_decks.is_empty() {
                        deck_status_msg =
                            "You have loaded no decks. Please import a deck to play!".to_string();
                    } else {
                        current_state = GameState::ModeSelect;
                    }
                }

                if draw_button(
                    "RETURN TO MENU",
                    screen_width() / 2.0,
                    bottom_y,
                    22,
                    mouse_pos,
                    text_primary,
                    text_muted,
                    accent_color,
                    custom_font.as_ref(),
                ) || is_key_pressed(KeyCode::Escape)
                {
                    current_state = GameState::Menu;
                }
            }

            GameState::ModeSelect => {
                draw_centered_text(
                    "MODE SELECTION",
                    65.0,
                    44,
                    text_primary,
                    custom_font.as_ref(),
                );

                // Center the 3 game mode boxes on the screen
                let box_size = 180.0;
                let spacing_x = 34.0;
                let total_w = box_size * 3.0 + spacing_x * 2.0;
                let start_x = screen_width() / 2.0 - total_w / 2.0;
                let box_y = screen_height() / 2.0 - box_size / 2.0 - 20.0;

                // Size the left Available Decks sidebar cleanly to fit left of the centered modes
                let left_panel_x = 28.0;
                let left_panel_y = 130.0;
                let left_panel_w = (start_x - left_panel_x - 28.0).clamp(210.0, 290.0);
                let left_panel_h = screen_height() - 200.0;

                let mut left_bg = text_muted;
                left_bg.a = 0.06;
                draw_rectangle(
                    left_panel_x,
                    left_panel_y,
                    left_panel_w,
                    left_panel_h,
                    left_bg,
                );
                draw_rectangle_lines(
                    left_panel_x,
                    left_panel_y,
                    left_panel_w,
                    left_panel_h,
                    1.5,
                    text_muted,
                );

                draw_ui_text(
                    "AVAILABLE DECKS",
                    left_panel_x + 14.0,
                    left_panel_y + 24.0,
                    17,
                    text_primary,
                    custom_font.as_ref(),
                );

                let count_label = format!("{} selected", selected_deck_names.len());
                draw_ui_text(
                    &count_label,
                    left_panel_x + 14.0,
                    left_panel_y + 42.0,
                    13,
                    accent_color,
                    custom_font.as_ref(),
                );

                let list_top = left_panel_y + 52.0;
                let list_area = Rect::new(
                    left_panel_x,
                    list_top,
                    left_panel_w,
                    left_panel_h - 52.0,
                );
                if list_area.contains(mouse_pos) {
                    let wheel = mouse_wheel().1;
                    mode_deck_scroll_offset = (mode_deck_scroll_offset - wheel * 20.0).clamp(
                        0.0,
                        (available_decks.len() as f32 * 36.0 - (left_panel_h - 72.0)).max(0.0),
                    );
                }

                let mut item_y_cursor = list_top + 22.0 - mode_deck_scroll_offset;
                for (i, deck_entry) in available_decks.iter().enumerate() {
                    if item_y_cursor > list_top + 6.0
                        && item_y_cursor < left_panel_y + left_panel_h - 16.0
                    {
                        let item_rect = Rect::new(
                            left_panel_x + 8.0,
                            item_y_cursor - 17.0,
                            left_panel_w - 16.0,
                            30.0,
                        );
                        let hovered = item_rect.contains(mouse_pos);
                        let is_checked = selected_deck_names.contains(&deck_entry.name);

                        if is_checked || hovered {
                            let mut hl = accent_color;
                            hl.a = if is_checked { 0.22 } else { 0.12 };
                            draw_rectangle(item_rect.x, item_rect.y, item_rect.w, item_rect.h, hl);
                        }

                        if hovered && is_mouse_button_pressed(MouseButton::Left) {
                            selected_deck_index = i;
                            if is_checked {
                                if selected_deck_names.len() > 1 {
                                    selected_deck_names.retain(|n| n != &deck_entry.name);
                                }
                            } else {
                                selected_deck_names.push(deck_entry.name.clone());
                            }
                        }

                        // Draw checkbox on the left of the deck row
                        let cb_size = 15.0;
                        let cb_x = item_rect.x + 8.0;
                        let cb_y = item_rect.y + (item_rect.h - cb_size) * 0.5;
                        draw_rectangle_lines(
                            cb_x,
                            cb_y,
                            cb_size,
                            cb_size,
                            1.6,
                            if is_checked || hovered {
                                accent_color
                            } else {
                                text_muted
                            },
                        );
                        if is_checked {
                            draw_rectangle(
                                cb_x + 3.0,
                                cb_y + 3.0,
                                cb_size - 6.0,
                                cb_size - 6.0,
                                accent_color,
                            );
                        }

                        // Draw deck name clipped cleanly to sidebar width
                        let text_x = cb_x + cb_size + 10.0;
                        let max_name_w = (item_rect.x + item_rect.w - text_x - 6.0).max(40.0);
                        let clipped_name = truncate_with_ellipsis(
                            &deck_entry.name,
                            max_name_w,
                            16,
                            custom_font.as_ref(),
                        );
                        draw_ui_text(
                            &clipped_name,
                            text_x,
                            item_y_cursor,
                            16,
                            if is_checked { text_primary } else { text_muted },
                            custom_font.as_ref(),
                        );
                    }
                    item_y_cursor += 36.0;
                }

                // Float Mode
                let float_rect = Rect::new(start_x, box_y, box_size, box_size);
                let float_hovered = float_rect.contains(mouse_pos);

                let mut f_box_fill = accent_color;
                f_box_fill.a = if float_hovered { 0.18 } else { 0.08 };
                draw_rectangle(
                    float_rect.x,
                    float_rect.y,
                    float_rect.w,
                    float_rect.h,
                    f_box_fill,
                );
                draw_rectangle_lines(
                    float_rect.x,
                    float_rect.y,
                    float_rect.w,
                    float_rect.h,
                    if float_hovered { 2.5 } else { 1.8 },
                    if float_hovered {
                        text_primary
                    } else {
                        accent_color
                    },
                );

                let alpha_mult = if float_hovered { 1.0 } else { 0.45 };
                for (idx, &ratio) in preview_bubbles_init.iter().enumerate() {
                    let cx = float_rect.x + float_rect.w * ratio.x;
                    let cy = float_rect.y + float_rect.h * ratio.y;
                    let r = 21.0;

                    let pos = if float_hovered {
                        let t = get_time() as f32 * 2.0 + idx as f32;
                        vec2(cx + t.cos() * 8.0, cy + t.sin() * 8.0)
                    } else {
                        vec2(cx, cy)
                    };

                    let mut bubble_fill = accent_color;
                    bubble_fill.a = 0.25 * alpha_mult;
                    let mut bubble_border = accent_color;
                    bubble_border.a = alpha_mult;

                    draw_circle(pos.x, pos.y, r, bubble_fill);
                    draw_circle_lines(pos.x, pos.y, r, 1.8, bubble_border);
                }

                let float_label = "Float Mode";
                let fl_dims = measure_text(float_label, custom_font.as_ref(), 20, 1.0);
                draw_ui_text(
                    float_label,
                    float_rect.x + float_rect.w / 2.0 - fl_dims.width / 2.0,
                    box_y + box_size + 28.0,
                    20,
                    if float_hovered { text_primary } else { text_muted },
                    custom_font.as_ref(),
                );

                if float_hovered && is_mouse_button_pressed(MouseButton::Left) {
                    available_decks = scan_available_decks();
                    if available_decks.is_empty() {
                        deck_status_msg =
                            "You have loaded no decks. Please import a deck to play!".to_string();
                        current_state = GameState::EditDeck;
                    } else {
                        current_state = GameState::FloatSetup;
                    }
                }

                // Tile Mode
                let tile_box_x = start_x + box_size + spacing_x;
                let tile_rect = Rect::new(tile_box_x, box_y, box_size, box_size);
                let tile_hovered = tile_rect.contains(mouse_pos);

                let mut t_box_fill = accent_color;
                t_box_fill.a = if tile_hovered { 0.18 } else { 0.08 };
                draw_rectangle(
                    tile_rect.x,
                    tile_rect.y,
                    tile_rect.w,
                    tile_rect.h,
                    t_box_fill,
                );
                draw_rectangle_lines(
                    tile_rect.x,
                    tile_rect.y,
                    tile_rect.w,
                    tile_rect.h,
                    if tile_hovered { 2.5 } else { 1.8 },
                    if tile_hovered {
                        text_primary
                    } else {
                        accent_color
                    },
                );

                let tile_alpha_mult = if tile_hovered { 1.0 } else { 0.45 };
                if tile_hovered {
                    tile_anim_timer += dt;
                    if tile_anim_timer >= 0.65 {
                        tile_anim_timer = 0.0;
                        if !tile_icon_cells.is_empty() {
                            let pop_idx = tile_pattern_idx % tile_icon_cells.len();
                            let old_cell = &tile_icon_cells[pop_idx];
                            let sub_grid_size = 4;
                            let sub_cell_size = box_size / sub_grid_size as f32;
                            let pop_center = vec2(
                                tile_rect.x + (old_cell.col as f32 + 0.5) * sub_cell_size,
                                tile_rect.y + (old_cell.row as f32 + 0.5) * sub_cell_size,
                            );
                            tile_icon_pops.push(PopEffect {
                                pos: pop_center,
                                base_radius: sub_cell_size * 0.42,
                                timer: 0.0,
                                duration: 0.32,
                                color: accent_color,
                            });

                            let mut next_loc =
                                tile_loop_pattern[tile_pattern_idx % tile_loop_pattern.len()];
                            for offset in 0..tile_loop_pattern.len() {
                                let cand = tile_loop_pattern
                                    [(tile_pattern_idx + offset) % tile_loop_pattern.len()];
                                if !tile_icon_cells
                                    .iter()
                                    .any(|c| c.row == cand.0 && c.col == cand.1)
                                {
                                    next_loc = cand;
                                    break;
                                }
                            }

                            tile_icon_cells[pop_idx] = TileIconCell {
                                row: next_loc.0,
                                col: next_loc.1,
                                scale: 0.05,

                                            &hint.display_text(),
                                            Rect::new(hint_x, hint_y, hint_panel_w, box_h),
                                            16,
                                            MIN_BOX_FONT_SIZE,
                                            text_primary,
                                            custom_font.as_ref(),
                                        );
                                    }
                                }
                            } else {
                                let prompt = format!(
                                    "Hint {} [Press {}]",
                                    i + 1,
                                    key_name(settings.hint_keys[i])
                                );
                                draw_ui_text(
                                    &prompt,
                                    hint_x + 16.0,
                                    hint_y + box_h / 2.0 + 6.0,
                                    16,
                                    text_muted,
                                    custom_font.as_ref(),
                                );
                            }

                            hint_y += box_h + 14.0;
                        }
                    }

                    let grid_rect = tile_grid_rect();
                    let cell_size = grid_rect.w / session.difficulty.grid_size as f32;

                    let mut zone_fill = text_muted;
                    zone_fill.a = if session.wrong_answer_triggered {
                        0.01
                    } else {
                        0.03
                    };
                    let mut zone_border = text_muted;
                    zone_border.a = if session.wrong_answer_triggered {
                        0.05
                    } else {
                        0.18
                    };
                    draw_rounded_box(
                        grid_rect.x,
                        grid_rect.y,
                        grid_rect.w,
                        grid_rect.h,
                        16.0,
                        zone_fill,
                        zone_border,
                    );

                    for target in &session.targets {
                        let r = target.rect(grid_rect.point(), cell_size);
                        let hovered = target.contains(mouse_pos, grid_rect.point(), cell_size);
                        let border_col = if hovered { text_primary } else { accent_color };
                        let mut fill_col = accent_color;
                        fill_col.a = if hovered { 0.28 } else { 0.16 };

                        draw_rounded_box(r.x, r.y, r.w, r.h, 10.0, fill_col, border_col);

                        let cx = r.x + r.w * 0.5;
                        let cy = r.y + r.h * 0.5;

                        match target.item.kind {
                            ContentType::Audio => {
                                draw_speaker_icon(
                                    cx,
                                    cy - r.h * 0.10,
                                    (r.w * 0.34).max(16.0),
                                    text_primary,
                                    target.hovered_audio,
                                );
                                let disp = target.item.display_text();
                                let f_size = ((14.0 * target.spawn_scale) as u16).max(MIN_BOX_FONT_SIZE);
                                let clipped = truncate_with_ellipsis(&disp, r.w - 14.0, f_size, custom_font.as_ref());
                                let dims = measure_text(&clipped, custom_font.as_ref(), f_size, 1.0);
                                draw_ui_text(
                                    &clipped,
                                    cx - dims.width / 2.0,
                                    cy + r.h * 0.34,
                                    f_size,
                                    if target.hovered_audio {
                                        accent_color
                                    } else {
                                        text_primary
                                    },
                                    custom_font.as_ref(),
                                );
                            }
                            ContentType::Image => {
                                if let Some(tex) = session.textures.get(&target.item.label) {
                                    let pad = (r.w * 0.12).max(4.0);
                                    let img_size = (r.w - pad * 2.0).max(8.0);
                                    draw_texture_ex(
                                        tex,
                                        r.x + pad,
                                        r.y + pad,
                                        WHITE,
                                        DrawTextureParams {
                                            dest_size: Some(vec2(img_size, img_size)),
                                            ..Default::default()
                                        },
                                    );
                                } else if target.spawn_scale > 0.35 {
                                    let disp = target.item.display_text();
                                    let pref = ((18.0 * target.spawn_scale) as u16)
                                        .clamp(MIN_BOX_FONT_SIZE, MAX_BOX_FONT_SIZE);
                                    draw_fitted_wrapped_text(
                                        &disp,
                                        r,
                                        pref,
                                        MIN_BOX_FONT_SIZE,
                                        text_primary,
                                        custom_font.as_ref(),
                                    );
                                }
                            }
                            ContentType::Text => {
                                if target.spawn_scale > 0.35 {
                                    let disp = target.item.display_text();
                                    let pref = ((19.0 * target.spawn_scale) as u16)
                                        .clamp(MIN_BOX_FONT_SIZE, MAX_BOX_FONT_SIZE);
                                    draw_fitted_wrapped_text(
                                        &disp,
                                        r,
                                        pref,
                                        MIN_BOX_FONT_SIZE,
                                        text_primary,
                                        custom_font.as_ref(),
                                    );
                                }
                            }
                        }
                    }

                    for pop in &session.pops {
                        let p = (pop.timer / pop.duration).clamp(0.0, 1.0);
                        let radius = pop.base_radius * (0.7 + 0.9 * p);
                        let mut c = pop.color;
                        c.a = (1.0 - p).powf(1.5);
                        draw_circle_lines(pop.pos.x, pop.pos.y, radius, 3.5 * (1.0 - p), c);
                    }
                }

                if round_finished || is_key_pressed(KeyCode::Escape) {
                    if let Some(session) = &mut tile_session {
                        session.stop_active_clip(&sounds);
                        if session.difficulty.is_highscore_eligible() {
                            let entry = deck_high_scores
                                .entry(session.deck_name.clone())
                                .or_insert(0);
                            if session.score > *entry {
                                *entry = session.score;
                            }
                        }
                    }
                    current_state = GameState::RoundSummary;
                }
            }

            GameState::DropPlaying => {
                let mut round_finished = false;

                if let Some(session) = &mut drop_session {
                    round_finished = session.update(dt, mouse_pos, &settings, &sounds);

                    if !session.wrong_answer_triggered {
                        draw_ui_text(
                            &format!("SCORE: {}", session.score),
                            24.0,
                            36.0,
                            28,
                            text_primary,
                            custom_font.as_ref(),
                        );

                        let time_str = if session.difficulty.infinite_round {
                            "ROUND TIME: INFINITE (ESC to Finish)".to_string()
                        } else {
                            let secs = session.round_time_left.ceil().max(0.0) as u32;
                            format!("ROUND TIME: {}:{:02}", secs / 60, secs % 60)
                        };
                        draw_ui_text(
                            &time_str,
                            24.0,
                            66.0,
                            18,
                            accent_color,
                            custom_font.as_ref(),
                        );

                        draw_ui_text(
                            &format!("DECK LOOPS: {}", session.loops_completed),
                            24.0,
                            92.0,
                            18,
                            text_muted,
                            custom_font.as_ref(),
                        );

                        draw_ui_text(
                            "SPACE: Skip Card (0 pts)",
                            24.0,
                            116.0,
                            15,
                            text_muted,
                            custom_font.as_ref(),
                        );
                    }

                    let current_card = &session.deck[session.current_question.card_idx];
                    let center_x = screen_width() / 2.0;
                    let q_box_size = 135.0;
                    let q_top_y = 28.0;

                    match current_card.question.kind {
                        ContentType::Audio => {
                            let bx = center_x - q_box_size / 2.0;
                            let mut fill = accent_color;
                            fill.a = 0.12;
                            draw_rectangle(bx, q_top_y, q_box_size, q_box_size, fill);
                            draw_rectangle_lines(
                                bx,
                                q_top_y,
                                q_box_size,
                                q_box_size,
                                2.5,
                                accent_color,
                            );
                            draw_speaker_icon(
                                center_x,
                                q_top_y + q_box_size * 0.42,
                                54.0,
                                text_primary,
                                session.audio_replay_pulse > 0.0,
                            );
                            draw_centered_text(
                                "Right-Click / SHIFT",
                                q_top_y + q_box_size - 12.0,
                                11,
                                text_muted,
                                custom_font.as_ref(),
                            );
                        }
                        ContentType::Image => {
                            let bx = center_x - q_box_size / 2.0;
                            let mut fill = accent_color;
                            fill.a = 0.12;
                            draw_rectangle(bx, q_top_y, q_box_size, q_box_size, fill);
                            draw_rectangle_lines(
                                bx,
                                q_top_y,
                                q_box_size,
                                q_box_size,
                                2.5,
                                accent_color,
                            );
                            if let Some(tex) = session.textures.get(&current_card.question.label) {
                                let pad = 8.0;
                                draw_texture_ex(
                                    tex,
                                    bx + pad,
                                    q_top_y + pad,
                                    WHITE,
                                    DrawTextureParams {
                                        dest_size: Some(vec2(
                                            q_box_size - pad * 2.0,
                                            q_box_size - pad * 2.0,
                                        )),
                                        ..Default::default()
                                    },
                                );
                            } else {
                                draw_fitted_wrapped_text(
                                    &current_card.question.display_text(),
                                    Rect::new(bx, q_top_y, q_box_size, q_box_size),
                                    20,
                                    MIN_BOX_FONT_SIZE,
                                    text_primary,
                                    custom_font.as_ref(),
                                );
                            }
                        }
                        ContentType::Text => {
                            let q_rect = Rect::new(center_x - 280.0, q_top_y, 560.0, 120.0);
                            draw_fitted_wrapped_text(
                                &current_card.question.display_text(),
                                q_rect,
                                28,
                                MIN_BOX_FONT_SIZE,
                                text_primary,
                                custom_font.as_ref(),
                            );
                        }
                    }

                    let playfield = drop_playfield_rect();
                    let num_lanes = session.difficulty.num_lanes.clamp(3, 7);
                    let lane_w = playfield.w / num_lanes as f32;
                    let hit_line_y = playfield.y + playfield.h - 46.0;

                    if !session.wrong_answer_triggered {
                        let hint_panel_w = 230.0;
                        let hint_x = screen_width() - hint_panel_w - 24.0;
                        let mut hint_y = 65.0;

                        draw_ui_text(
                            "HINTS",
                            hint_x,
                            hint_y,
                            20,
                            text_primary,
                            custom_font.as_ref(),
                        );
                        hint_y += 16.0;

                        for (i, hint) in current_card.hints.iter().enumerate().take(3) {
                            let box_h = 72.0;
                            let mut fill = accent_color;
                            fill.a = if session.hints_revealed[i] {
                                0.16
                            } else {
                                0.05
                            };
                            let border = if session.hints_revealed[i] {
                                accent_color
                            } else {
                                text_muted
                            };

                            draw_rounded_box(hint_x, hint_y, hint_panel_w, box_h, 10.0, fill, border);

                            if session.hints_revealed[i] {
                                match hint.kind {
                                    ContentType::Audio => {
                                        draw_speaker_icon(
                                            hint_x + 36.0,
                                            hint_y + box_h / 2.0,
                                            34.0,
                                            accent_color,
                                            session.hint_audio_pulse[i] > 0.0,
                                        );
                                        draw_fitted_wrapped_text(
                                            &hint.display_text(),
                                            Rect::new(hint_x + 62.0, hint_y, hint_panel_w - 68.0, box_h),
                                            16,
                                            MIN_BOX_FONT_SIZE,
                                            text_primary,
                                            custom_font.as_ref(),
                                        );
                                    }
                                    ContentType::Image => {
                                        if let Some(tex) = session.textures.get(&hint.label) {
                                            draw_texture_ex(
                                                tex,
                                                hint_x + 14.0,
                                                hint_y + 10.0,
                                                WHITE,
                                                DrawTextureParams {
                                                    dest_size: Some(vec2(52.0, 52.0)),
                                                    ..Default::default()
                                                },
                                            );
                                        } else {
                                            draw_rectangle_lines(
                                                hint_x + 14.0,
                                                hint_y + 10.0,
                                                52.0,
                                                52.0,
                                                1.5,
                                                accent_color,
                                            );
                                        }
                                        draw_fitted_wrapped_text(
                                            &hint.display_text(),
                                            Rect::new(hint_x + 70.0, hint_y, hint_panel_w - 76.0, box_h),
                                            16,
                                            MIN_BOX_FONT_SIZE,
                                            text_primary,
                                            custom_font.as_ref(),
                                        );
                                    }
                                    ContentType::Text => {
                                        draw_fitted_wrapped_text(
                                            &hint.display_text(),
                                            Rect::new(hint_x, hint_y, hint_panel_w, box_h),
                                            16,
                                            MIN_BOX_FONT_SIZE,
                                            text_primary,
                                            custom_font.as_ref(),
                                        );
                                    }
                                }
                            } else {
                                let prompt = format!(
                                    "Hint {} [Press {}]",
                                    i + 1,
                                    key_name(settings.hint_keys[i])
                                );
                                draw_ui_text(
                                    &prompt,
                                    hint_x + 16.0,
                                    hint_y + box_h / 2.0 + 6.0,
                                    16,
                                    text_muted,
                                    custom_font.as_ref(),
                                );
                            }

                            hint_y += box_h + 14.0;
                        }

                        let mut zone_fill = text_muted;
                        zone_fill.a = 0.025;
                        let mut zone_border = text_muted;
                        zone_border.a = 0.18;
                        draw_rounded_box(
                            playfield.x,
                            playfield.y,
                            playfield.w,
                            playfield.h,
                            14.0,
                            zone_fill,
                            zone_border,
                        );

                        for lane in 0..num_lanes {
                            let lx = playfield.x + lane as f32 * lane_w;
                            let grad_steps = 18;
                            let step_h = (hit_line_y - playfield.y) / grad_steps as f32;
                            for s in 0..grad_steps {
                                let frac = s as f32 / grad_steps as f32;
                                let mut col = accent_color;
                                col.a = 0.01 + 0.11 * frac.powf(2.2);
                                draw_rectangle(
                                    lx + 2.0,
                                    playfield.y + s as f32 * step_h,
                                    lane_w - 4.0,
                                    step_h + 0.5,
                                    col,
                                );
                            }

                            if lane > 0 {
                                let mut div_col = text_muted;
                                div_col.a = 0.16;
                                draw_line(
                                    lx,
                                    playfield.y + 8.0,
                                    lx,
                                    hit_line_y,
                                    1.5,
                                    div_col,
                                );
                            }
                        }

                        let mut glow_col = accent_color;
                        glow_col.a = 0.22;
                        draw_rectangle(
                            playfield.x + 4.0,
                            hit_line_y - 6.0,
                            playfield.w - 8.0,
                            12.0,
                            glow_col,
                        );
                        draw_line(
                            playfield.x + 4.0,
                            hit_line_y,
                            playfield.x + playfield.w - 4.0,
                            hit_line_y,
                            3.0,
                            accent_color,
                        );
                    }

                    for tile in &session.tiles {
                        let r = tile.rect(playfield, num_lanes);
                        let hovered = tile.contains(mouse_pos, playfield, num_lanes);
                        let border_col = if hovered { text_primary } else { accent_color };
                        let mut fill_col = accent_color;
                        fill_col.a = if session.wrong_answer_triggered {
                            if hovered { 0.34 } else { 0.24 }
                        } else if hovered {
                            0.30
                        } else {
                            0.18
                        };

                        draw_rounded_box(r.x, r.y, r.w, r.h, 10.0, fill_col, border_col);

                        let cx = r.x + r.w * 0.5;
                        let cy = r.y + r.h * 0.5;

                        match tile.item.kind {
                            ContentType::Audio => {
                                draw_speaker_icon(
                                    cx,
                                    cy - r.h * 0.12,
                                    (r.h * 0.42).max(18.0),
                                    text_primary,
                                    tile.hovered_audio,
                                );
                                let disp = tile.item.display_text();
                                let clipped = truncate_with_ellipsis(
                                    &disp,
                                    r.w - 14.0,
                                    MIN_BOX_FONT_SIZE,
                                    custom_font.as_ref(),
                                );
                                let dims =
                                    measure_text(&clipped, custom_font.as_ref(), MIN_BOX_FONT_SIZE, 1.0);
                                draw_ui_text(
                                    &clipped,
                                    cx - dims.width / 2.0,
                                    cy + r.h * 0.34,
                                    MIN_BOX_FONT_SIZE,
                                    if tile.hovered_audio {
                                        accent_color
                                    } else {
                                        text_primary
                                    },
                                    custom_font.as_ref(),
                                );
                            }
                            ContentType::Image => {
                                if let Some(tex) = session.textures.get(&tile.item.label) {
                                    let img_size = (r.h - 14.0).min(r.w - 14.0).max(16.0);
                                    draw_texture_ex(
                                        tex,
                                        cx - img_size / 2.0,
                                        cy - img_size / 2.0,
                                        WHITE,
                                        DrawTextureParams {
                                            dest_size: Some(vec2(img_size, img_size)),
                                            ..Default::default()
                                        },
                                    );
                                } else {
                                    let disp = tile.item.display_text();
                                    draw_fitted_wrapped_text(
                                        &disp,
                                        r,
                                        18,
                                        MIN_BOX_FONT_SIZE,
                                        text_primary,
                                        custom_font.as_ref(),
                                    );
                                }
                            }
                            ContentType::Text => {
                                let disp = tile.item.display_text();
                                draw_fitted_wrapped_text(
                                    &disp,
                                    r,
                                    19,
                                    MIN_BOX_FONT_SIZE,
                                    text_primary,
                                    custom_font.as_ref(),
                                );
                            }
                        }
                    }

                    if session.wrong_answer_triggered {
                        draw_centered_text(
                            "Incorrect! Click the highlighted correct answer to continue",
                            hit_line_y + 32.0,
                            18,
                            accent_color,
                            custom_font.as_ref(),
                        );
                    }

                    for pop in &session.pops {
                        let p = (pop.timer / pop.duration).clamp(0.0, 1.0);
                        let radius = pop.base_radius * (0.7 + 0.9 * p);
                        let mut c = pop.color;
                        c.a = (1.0 - p).powf(1.5);
                        draw_circle_lines(pop.pos.x, pop.pos.y, radius, 3.5 * (1.0 - p), c);
                    }
                }

                if round_finished || is_key_pressed(KeyCode::Escape) {
                    if let Some(session) = &mut drop_session {
                        session.stop_active_clip(&sounds);
                        if session.difficulty.is_highscore_eligible() {
                            let entry = deck_high_scores
                                .entry(session.deck_name.clone())
                                .or_insert(0);
                            if session.score > *entry {
                                *entry = session.score;
                            }
                        }
                    }
                    current_state = GameState::RoundSummary;
                }
            }

            GameState::RoundSummary => {
                draw_centered_text(
                    "ROUND SUMMARY",
                    120.0,
                    46,
                    text_primary,
                    custom_font.as_ref(),
                );

                let score_val = if let Some(s) = &float_session {
                    s.score
                } else if let Some(s) = &tile_session {
                    s.score
                } else if let Some(s) = &drop_session {
                    s.score
                } else {
                    0
                };
                let deck_nm = if let Some(s) = &float_session {
                    s.deck_name.clone()
                } else if let Some(s) = &tile_session {
                    s.deck_name.clone()
                } else if let Some(s) = &drop_session {
                    s.deck_name.clone()
                } else {
                    "Unknown".to_string()
                };
                let loops_val = if let Some(s) = &float_session {
                    s.loops_completed
                } else if let Some(s) = &tile_session {
                    s.loops_completed
                } else if let Some(s) = &drop_session {
                    s.loops_completed
                } else {
                    0
                };

                draw_centered_text(
                    &format!("Deck: {}", deck_nm),
                    170.0,
                    20,
                    text_muted,
                    custom_font.as_ref(),
                );

                draw_centered_text(
                    &format!("FINAL SCORE: {}", score_val),
                    245.0,
                    36,
                    accent_color,
                    custom_font.as_ref(),
                );

                let hs_best = deck_high_scores.get(&deck_nm).copied().unwrap_or(score_val);
                draw_centered_text(
                    &format!("HIGH SCORE (1:00 Standard): {}", hs_best),
                    305.0,
                    22,
                    text_primary,
                    custom_font.as_ref(),
                );

                draw_centered_text(
                    &format!("DECK LOOPS COMPLETED: {}", loops_val),
                    355.0,
                    22,
                    text_primary,
                    custom_font.as_ref(),
                );

                let bottom_y = screen_height() - 65.0;
                if draw_button(
                    "REPLAY",
                    screen_width() / 2.0,
                    bottom_y - 55.0,
                    26,
                    mouse_pos,
                    text_primary,
                    text_muted,
                    accent_color,
                    custom_font.as_ref(),
                ) {
                    let active_decks: Vec<DeckEntry> = available_decks
                        .iter()
                        .filter(|d| selected_deck_names.contains(&d.name))
                        .cloned()
                        .collect();
                    if float_session.is_some() {
                        float_session = Some(
                            FloatSession::new(&active_decks, float_difficulty.clone(), &sounds)
                                .await,
                        );
                        current_state = GameState::FloatPlaying;
                    } else if tile_session.is_some() {
                        tile_session = Some(
                            TileSession::new(&active_decks, tile_difficulty.clone(), &sounds).await,
                        );
                        current_state = GameState::TilePlaying;
                    } else if drop_session.is_some() {
                        drop_session = Some(
                            DropSession::new(&active_decks, drop_difficulty.clone(), &sounds).await,
                        );
                        current_state = GameState::DropPlaying;
                    }
                }

                if draw_button(
                    "RETURN TO MAIN MENU",
                    screen_width() / 2.0,
                    bottom_y,
                    22,
                    mouse_pos,
                    text_primary,
                    text_muted,
                    accent_color,
                    custom_font.as_ref(),
                ) {
                    current_state = GameState::Menu;
                }
            }

            GameState::Settings => {
                draw_centered_text("SETTINGS", 40.0, 34, text_primary, custom_font.as_ref());

                let panel_w = 560.0_f32.min(screen_width() - 60.0);
                let panel_x = screen_width() / 2.0 - panel_w / 2.0;

                settings.dot_radius = draw_slider(
                    "Cursor Size",
                    &format!("{:.1} px", settings.dot_radius),
                    settings.dot_radius,
                    MIN_DOT_RADIUS,
                    MAX_DOT_RADIUS,
                    panel_x,
                    74.0,
                    panel_w,
                    mouse_pos,
                    &settings,
                    custom_font.as_ref(),
                );

                settings.trail_radius_mult = draw_slider(
                    "Trail Radius Multiplier",
                    &format!("{:.1}x", settings.trail_radius_mult),
                    settings.trail_radius_mult,
                    MIN_TRAIL_MULT,
                    MAX_TRAIL_MULT,
                    panel_x,
                    116.0,
                    panel_w,
                    mouse_pos,
                    &settings,
                    custom_font.as_ref(),
                );

                settings.max_trail_length = draw_slider(
                    "Max Trail Length (1 segment/px)",
                    &format!("{} px", settings.max_trail_length.round() as u32),
                    settings.max_trail_length,
                    MIN_TRAIL_LENGTH,
                    MAX_TRAIL_LENGTH_CAP,
                    panel_x,
                    158.0,
                    panel_w,
                    mouse_pos,
                    &settings,
                    custom_font.as_ref(),
                )
                .round();

                let display_y = 204.0;
                let disp_rect = Rect::new(panel_x, display_y - 16.0, panel_w, 24.0);
                let disp_hover = disp_rect.contains(mouse_pos);
                if disp_hover {
                    let mut hl = accent_color;
                    hl.a = 0.12;
                    draw_rectangle(
                        disp_rect.x - 6.0,
                        disp_rect.y,
                        disp_rect.w + 12.0,
                        disp_rect.h,
                        hl,
                    );
                }
                draw_ui_text(
                    "Display Mode",
                    panel_x,
                    display_y,
                    17,
                    text_primary,
                    custom_font.as_ref(),
                );
                let mode_str = format!("[ {} ]", settings.display_mode.label());
                let mode_dims = measure_text(&mode_str, custom_font.as_ref(), 17, 1.0);
                draw_ui_text(
                    &mode_str,
                    panel_x + panel_w - mode_dims.width,
                    display_y,
                    17,
                    accent_color,
                    custom_font.as_ref(),
                );
                if disp_hover && is_mouse_button_pressed(MouseButton::Left) {
                    settings.display_mode = settings.display_mode.next();
                    settings.display_mode.apply();
                }

                let fps_y = 230.0;
                let fps_rect = Rect::new(panel_x, fps_y - 16.0, panel_w, 24.0);
                let fps_hover = fps_rect.contains(mouse_pos);
                if fps_hover {
                    let mut hl = accent_color;
                    hl.a = 0.12;
                    draw_rectangle(
                        fps_rect.x - 6.0,
                        fps_rect.y,
                        fps_rect.w + 12.0,
                        fps_rect.h,
                        hl,
                    );
                }
                draw_ui_text(
                    "Show FPS Counter",
                    panel_x,
                    fps_y,
                    17,
                    text_primary,
                    custom_font.as_ref(),
                );
                let fps_str = if settings.show_fps {
                    "[ ON ]"
                } else {
                    "[ OFF ]"
                };
                let fps_dims = measure_text(fps_str, custom_font.as_ref(), 17, 1.0);
                draw_ui_text(
                    fps_str,
                    panel_x + panel_w - fps_dims.width,
                    fps_y,
                    17,
                    accent_color,
                    custom_font.as_ref(),
                );
                if fps_hover && is_mouse_button_pressed(MouseButton::Left) {
                    settings.show_fps = !settings.show_fps;
                }

                draw_ui_text(
                    "CONTROLS (Click a row to rebind Hint keys)",
                    panel_x,
                    264.0,
                    15,
                    text_muted,
                    custom_font.as_ref(),
                );

                if let Some(bind_idx) = rebinding_hint_idx {
                    if let Some(pressed_key) = get_last_key_pressed() {
                        if pressed_key != KeyCode::Escape {
                            settings.hint_keys[bind_idx] = pressed_key;
                        }
                        rebinding_hint_idx = None;
                    }
                }

                let mut ctrl_y = 288.0;
                for i in 0..3 {
                    let r_rect = Rect::new(panel_x, ctrl_y - 16.0, panel_w, 24.0);
                    let r_hover = r_rect.contains(mouse_pos);
                    if r_hover {
                        let mut hl = accent_color;
                        hl.a = 0.12;
                        draw_rectangle(r_rect.x - 6.0, r_rect.y, r_rect.w + 12.0, r_rect.h, hl);
                    }

                    let row_title = format!(
                        "Reveal Hint {} ({} time left)",
                        i + 1,
                        HINT_THRESHOLDS_LABEL[i]
                    );
                    draw_ui_text(
                        &row_title,
                        panel_x,
                        ctrl_y,
                        17,
                        text_primary,
                        custom_font.as_ref(),
                    );

                    let key_str = if rebinding_hint_idx == Some(i) {
                        "[ PRESS ANY KEY... ]".to_string()
                    } else {
                        format!("[ Key: {} ]", key_name(settings.hint_keys[i]))
                    };
                    let k_dims = measure_text(&key_str, custom_font.as_ref(), 17, 1.0);
                    draw_ui_text(
                        &key_str,
                        panel_x + panel_w - k_dims.width,
                        ctrl_y,
                        17,
                        accent_color,
                        custom_font.as_ref(),
                    );

                    if r_hover && is_mouse_button_pressed(MouseButton::Left) {
                        rebinding_hint_idx = Some(i);
                        active_hex_slot = None;
                    }

                    ctrl_y += 26.0;
                }

                draw_ui_text(
                    "THEME COLORS (Click swatch to cycle, or click hex to type)",
                    panel_x,
                    376.0,
                    15,
                    text_muted,
                    custom_font.as_ref(),
                );

                if let Some(slot) = active_hex_slot {
                    while let Some(c) = get_char_pressed() {
                        if c.is_ascii_hexdigit() && hex_input_buffer.len() < 6 {
                            hex_input_buffer.push(c.to_ascii_uppercase());
                            if hex_input_buffer.len() == 6 {
                                if let Ok(parsed) = u32::from_str_radix(&hex_input_buffer, 16) {
                                    settings.set_hex(slot, parsed);
                                }
                                active_hex_slot = None;
                            }
                        }
                    }
                    if is_key_pressed(KeyCode::Backspace) {
                        hex_input_buffer.pop();
                    }
                    if is_key_pressed(KeyCode::Enter) {
                        if let Ok(parsed) = u32::from_str_radix(&hex_input_buffer, 16) {
                            settings.set_hex(slot, parsed);
                        }
                        active_hex_slot = None;
                    }
                }

                let mut row_y = 402.0;
                let swatch_x_fixed = panel_x + panel_w - 180.0;
                let hex_x_fixed = panel_x + panel_w - 140.0;

                for (label, slot) in COLOR_SLOTS {
                    let row_rect = Rect::new(panel_x, row_y - 16.0, panel_w, 24.0);
                    let hovered = row_rect.contains(mouse_pos);

                    if hovered {
                        let mut hl = accent_color;
                        hl.a = 0.12;
                        draw_rectangle(
                            row_rect.x - 6.0,
                            row_rect.y,
                            row_rect.w + 12.0,
                            row_rect.h,
                            hl,
                        );
                    }

                    draw_ui_text(label, panel_x, row_y, 17, text_primary, custom_font.as_ref());

                    let hex_val = settings.get_hex(*slot);
                    let is_editing = active_hex_slot == Some(*slot);
                    let hex_str = if is_editing {
                        format!("#{}|", hex_input_buffer)
                    } else {
                        format!("#{:06X}", hex_val)
                    };

                    draw_rectangle(
                        swatch_x_fixed,
                        row_y - 14.0,
                        16.0,
                        16.0,
                        Color::from_hex(hex_val),
                    );
                    draw_rectangle_lines(
                        swatch_x_fixed,
                        row_y - 14.0,
                        16.0,
                        16.0,
                        1.5,
                        text_primary,
                    );

                    let hex_color = if is_editing { accent_color } else { text_muted };
                    draw_ui_text(
                        &hex_str,
                        hex_x_fixed,
                        row_y,
                        17,
                        hex_color,
                        custom_font.as_ref(),
                    );

                    if hovered && is_mouse_button_pressed(MouseButton::Left) {
                        rebinding_hint_idx = None;
                        if mouse_pos.x >= hex_x_fixed - 10.0 {
                            active_hex_slot = Some(*slot);
                            hex_input_buffer.clear();
                        } else {
                            active_hex_slot = None;
                            settings.cycle_preset(*slot);
                        }
                    }

                    row_y += 27.0;
                }

                let btn_y = row_y + 18.0;

                let reset_label = "[ RESET DEFAULTS ]";
                let reset_dims = measure_text(reset_label, custom_font.as_ref(), 20, 1.0);
                let reset_rect = Rect::new(panel_x, btn_y - 20.0, reset_dims.width, 28.0);
                let reset_hover = reset_rect.contains(mouse_pos);
                draw_ui_text(
                    reset_label,
                    panel_x,
                    btn_y,
                    20,
                    if reset_hover {
                        accent_color
                    } else {
                        text_muted
                    },
                    custom_font.as_ref(),
                );
                if reset_hover && is_mouse_button_pressed(MouseButton::Left) {
                    settings = GameSettings::default();
                    settings.display_mode.apply();
                    active_hex_slot = None;
                    rebinding_hint_idx = None;
                }

                let back_label = "[ BACK (ESC) ]";
                let back_dims = measure_text(back_label, custom_font.as_ref(), 20, 1.0);
                let back_x = panel_x + panel_w - back_dims.width;
                let back_rect = Rect::new(back_x, btn_y - 20.0, back_dims.width, 28.0);
                let back_hover = back_rect.contains(mouse_pos);
                draw_ui_text(
                    back_label,
                    back_x,
                    btn_y,
                    20,
                    if back_hover {
                        text_primary
                    } else {
                        accent_color
                    },
                    custom_font.as_ref(),
                );
                if (back_hover && is_mouse_button_pressed(MouseButton::Left))
                    || (is_key_pressed(KeyCode::Escape) && rebinding_hint_idx.is_none())
                {
                    active_hex_slot = None;
                    rebinding_hint_idx = None;
                    current_state = GameState::Menu;
                }
            }
        }

        if settings.show_fps {
            let fps_text = format!("{} FPS", get_fps());
            let fps_dims = measure_text(&fps_text, custom_font.as_ref(), 16, 1.0);
            draw_ui_text(
                &fps_text,
                screen_width() - fps_dims.width - 14.0,
                22.0,
                16,
                accent_color,
                custom_font.as_ref(),
            );
        }

        custom_cursor.update_and_draw(dt, &settings);

        next_frame().await;
    }
}
